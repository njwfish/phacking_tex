\begin{abstract}
Specification search over inverse-probability-weighted and augmented
inverse-probability-weighted estimators is a search over fitted nuisance
models as well as reported targets.  A support-indexed estimating equation
gives a common linear--quadratic expansion for both estimators.  Under
random assignment and linear potential outcomes, logistic IPW and AIPW with
one joint support have the same leading search field.  Its coordinate-search
maximum is bounded at scale
$\Psi_s\sqrt{s\log(ep/s)/n}+\sigma_Ys\log(ep/s)/n$, where $\Psi_s$ measures
sparse prognostic signal and $\sigma_Y$ is the outcome scale.  Search over
irrelevant Gaussian covariates has a sharper product form.  At balance,
outcome regression and AIPW have the same residual-noise field, while IPW
also searches over predictable outcome variation.  In a planted
Gaussian--logistic confounding model, all three methods have the same local
omitted-variable bias, determined by the conditional covariance alignment
of the omitted propensity and outcome directions.  Search over arbitrary
covariate subspaces replaces $s\log(ep/s)$ by $p$ and admits an exact rank
reduction: one fitted direction for IPW and at most two joint nuisance
directions for AIPW.
\end{abstract}

\section{A support-indexed M-estimator}

Let $O_1,\ldots,O_n$ be independent observations with law $P$, and write
$P_n f=n^{-1}\sum_{i=1}^n f(O_i)$ and
$\mathbb G_n f=\sqrt n(P_n-P)f$.  Candidate covariates are indexed by
$[p]$, and the researcher may fit any support $S$ in
\begin{equation*}
        \mathcal S_s=\{S\subseteq[p]:|S|\le s\}.
\end{equation*}
For each $S$, let $\eta_S\in\R^{d_S}$ collect the nuisance parameters and
let $m_S(O;\eta_S)\in\R^{d_S}$ be their estimating function.  Its
population root $\eta_S^\star$ and empirical root $\hat\eta_S$ satisfy
\begin{equation}\label{eq:nuisance-roots}
        Pm_S(O;\eta_S^\star)=0,
        \qquad
        P_nm_S(O;\hat\eta_S)=0.
\end{equation}
The support-specific target is the mean of a scalar estimating map
$q_S(O;\eta_S)$,
\begin{equation}\label{eq:target-definition}
        \theta_S^\star=Pq_S(O;\eta_S^\star),
        \qquad
        \hat\theta_S=P_nq_S(O;\hat\eta_S).
\end{equation}
This form includes a joint Z-estimator because $q_S(O;\eta_S)-\theta$ is
the estimating equation for $\theta$.

Write a dot for differentiation with respect to $\eta_S$ and set
\begin{equation}\label{eq:population-derivatives}
        H_S=P\dot m_S(O;\eta_S^\star),
        \qquad
        a_S=P\dot q_S(O;\eta_S^\star).
\end{equation}
Here $H_S$ is a $d_S\times d_S$ matrix and $a_S$ is a $d_S$-vector.  The
following deterministic expansion isolates the only first-order object
needed for all supports.

\begin{proposition}[Uniform M-estimation expansion]
\label{prop:first-order}
Suppose every $H_S$ is invertible and, for a constant $\kappa$,
\begin{equation*}
        \max_{S\in\mathcal S_s}\norm{H_S^{-1}}_{\mathrm{op}}\le\kappa.
\end{equation*}
On an event on which all empirical roots in \eqref{eq:nuisance-roots}
exist, put
\begin{align*}
 r_n&=\max_{S\in\mathcal S_s}\norm{\hat\eta_S-\eta_S^\star},\\
 \epsilon_{m,n}
 &=\max_{S\in\mathcal S_s}\sup_{0\le t\le1}
 \norm{P_n\dot m_S\{O;\eta_S^\star
 +t(\hat\eta_S-\eta_S^\star)\}-H_S}_{\mathrm{op}},\\
 \epsilon_{q,n}
 &=\max_{S\in\mathcal S_s}\sup_{0\le t\le1}
 \norm{P_n\dot q_S\{O;\eta_S^\star
 +t(\hat\eta_S-\eta_S^\star)\}-a_S}.
\end{align*}
Then, simultaneously over $S\in\mathcal S_s$,
\begin{equation}\label{eq:nuisance-first-order}
 \hat\eta_S-\eta_S^\star
 =-\frac1{\sqrt n}H_S^{-1}\mathbb G_nm_S(O;\eta_S^\star)
 +R_{\eta,S},
 \qquad
 \norm{R_{\eta,S}}\le\kappa\epsilon_{m,n}r_n,
\end{equation}
and
\begin{equation}\label{eq:target-first-order}
 \hat\theta_S-\theta_S^\star
 =\frac1{\sqrt n}\mathbb G_n\varphi_S+R_{\theta,S},
 \qquad
 \varphi_S
 =q_S(O;\eta_S^\star)-\theta_S^\star
 -a_S^\top H_S^{-1}m_S(O;\eta_S^\star),
\end{equation}
where
\begin{equation}\label{eq:first-order-remainder}
 |R_{\theta,S}|
 \le
 \bigl(\max_{T\in\mathcal S_s}\norm{a_T}\,
       \kappa\epsilon_{m,n}+\epsilon_{q,n}\bigr)r_n.
\end{equation}
\end{proposition}

\begin{proof}
The fundamental theorem of calculus and the empirical equation give
\begin{equation*}
 0=\frac1{\sqrt n}\mathbb G_nm_S(O;\eta_S^\star)
   +H_S(\hat\eta_S-\eta_S^\star)
   +\int_0^1\{P_n\dot m_S(O;\eta_{S,t})-H_S\}
      (\hat\eta_S-\eta_S^\star)\,dt,
\end{equation*}
where $\eta_{S,t}=\eta_S^\star+t(\hat\eta_S-\eta_S^\star)$.
Multiplication by $H_S^{-1}$ proves \eqref{eq:nuisance-first-order} and
its remainder bound.  Applying the same identity to $q_S$ yields
\begin{equation*}
 \hat\theta_S-\theta_S^\star
 =\frac1{\sqrt n}\mathbb G_nq_S(O;\eta_S^\star)
  +a_S^\top(\hat\eta_S-\eta_S^\star)+R_{q,S},
 \qquad |R_{q,S}|\le\epsilon_{q,n}r_n.
\end{equation*}
Substitution of \eqref{eq:nuisance-first-order} proves
\eqref{eq:target-first-order}--\eqref{eq:first-order-remainder}.
\end{proof}

The population derivative $a_S$ determines whether fitted nuisances enter
the target at first order.  When $a_S=0$, Proposition~\ref{prop:first-order}
reduces the leading term to $\mathbb G_n
q_S(O;\eta_S^\star)/\sqrt n$.  If this function is also common across the
searched supports, its empirical fluctuation cancels from every searched
contrast.  The first support-dependent term is then quadratic.

\section{The quadratic Taylor field}

The quadratic term is the analogue of the covariance-aware field for
linear-regression search.  It can be written without choosing a particular
nuisance model.  Let $\ddot m_S[u,u]$ denote the vector obtained by applying
the Hessian of each component of $m_S$ to $(u,u)$, and use the same notation
for the scalar Hessian of $q_S$.  Define
\begin{equation}\label{eq:u-definition}
        u_S=H_S^{-1}\mathbb G_nm_S(O;\eta_S^\star).
\end{equation}

\begin{proposition}[Second-order expansion]
\label{prop:second-order}
Suppose the maps $m_S$ and $q_S$ are three times continuously
differentiable on neighborhoods of their population roots.  Let $c_n\ge1$
satisfy $c_n/\sqrt n=o(1)$.  Use a star on a function or derivative to
mean evaluation at $\eta_S^\star$.  Assume uniformly over
$S\in\mathcal S_s$ that the inverse Jacobians and population derivatives
through order three are bounded.  Assume also that the empirical third
derivatives are $O_{\Prob}(1)$ on the line segment from $\eta_S^\star$ to
$\hat\eta_S$ and that
\begin{equation}\label{eq:second-order-field-size}
 \norm{\mathbb G_nm_S^\star}
 +|\mathbb G_nq_S^\star|
 +\norm{\mathbb G_n\dot m_S^\star}_{\mathrm{op}}
 +\norm{\mathbb G_n\dot q_S^\star}
 +\norm{\mathbb G_n\ddot m_S^\star}
 +\norm{\mathbb G_n\ddot q_S^\star}
 =O_{\Prob}(c_n),
\end{equation}
and that $\max_S\norm{\hat\eta_S-\eta_S^\star}
=O_{\Prob}(c_n/\sqrt n)$.  Then, uniformly over $S$,
\begin{equation}\label{eq:nuisance-second-order}
\begin{split}
 \hat\eta_S-\eta_S^\star
 ={}&-\frac{u_S}{\sqrt n}
 +\frac1nH_S^{-1}\left\{
       \mathbb G_n\dot m_S^\star u_S
       -\frac12P\ddot m_S^\star[u_S,u_S]
       \right\}
 +O_{\Prob}\left(\frac{c_n^3}{n^{3/2}}\right),
\end{split}
\end{equation}
and
\begin{equation}\label{eq:target-second-order}
 \hat\theta_S-\theta_S^\star
 =\frac{L_S}{\sqrt n}+\frac{Q_S}{n}
 +O_{\Prob}\left(\frac{c_n^3}{n^{3/2}}\right),
\end{equation}
where
\begin{align}
 L_S={}&\mathbb G_nq_S^\star-a_S^\top u_S,
 \label{eq:linear-field}\\
 Q_S={}&a_S^\top H_S^{-1}\left\{
       \mathbb G_n\dot m_S^\star u_S
       -\frac12P\ddot m_S^\star[u_S,u_S]
       \right\}
       -(\mathbb G_n\dot q_S^\star)^\top u_S
       +\frac12P\ddot q_S^\star[u_S,u_S].
 \label{eq:quadratic-field}
\end{align}
\end{proposition}

\begin{proof}
Put $\Delta_S=\hat\eta_S-\eta_S^\star$.  Taylor expansion of the empirical
nuisance equation around $\eta_S^\star$ gives
\begin{equation*}
 0=\frac1{\sqrt n}\mathbb G_nm_S^\star
   +H_S\Delta_S
   +\frac1{\sqrt n}\mathbb G_n\dot m_S^\star\Delta_S
   +\frac12P\ddot m_S^\star[\Delta_S,\Delta_S]
   +O_{\Prob}\left(\frac{c_n^3}{n^{3/2}}\right),
\end{equation*}
uniformly in $S$.
The empirical Hessian fluctuation and the third-order Taylor term are of
the displayed remainder order under \eqref{eq:second-order-field-size}.
Define
\begin{equation*}
 v_S=H_S^{-1}\left\{
       \mathbb G_n\dot m_S^\star u_S
       -\frac12P\ddot m_S^\star[u_S,u_S]\right\}.
\end{equation*}
Substitution of $-u_S/\sqrt n+v_S/n$ into the preceding display cancels
its terms of orders $n^{-1/2}$ and $n^{-1}$.  The remaining terms are
$O_{\Prob}(c_n^3/n^{3/2})$ uniformly in $S$.  The empirical Jacobian on
the segment joining $\eta_S^\star$ and $\hat\eta_S$ is
$H_S+o_{\Prob}(1)$ uniformly in $S$, so the mean value identity and
boundedness of $H_S^{-1}$ give \eqref{eq:nuisance-second-order}.  A Taylor
expansion of the target yields
\begin{equation*}
 \hat\theta_S-\theta_S^\star
 =\frac1{\sqrt n}\mathbb G_nq_S^\star
  +a_S^\top\Delta_S
  +\frac1{\sqrt n}(\mathbb G_n\dot q_S^\star)^\top\Delta_S
  +\frac12P\ddot q_S^\star[\Delta_S,\Delta_S]
  +O_{\Prob}\left(\frac{c_n^3}{n^{3/2}}\right).
\end{equation*}
Substitution of \eqref{eq:nuisance-second-order} proves
\eqref{eq:target-second-order}--\eqref{eq:quadratic-field}.
\end{proof}

The expansion depends on the data only through the joint empirical field
\begin{equation}\label{eq:joint-empirical-field}
 \left\{\mathbb G_nq_S^\star,\,
 \mathbb G_nm_S^\star,\,
 \mathbb G_n\dot q_S^\star,\,
 \mathbb G_n\dot m_S^\star:S\in\mathcal S_s\right\}.
\end{equation}
Replacing this field by a centered Gaussian field with the same covariance
turns \eqref{eq:linear-field}--\eqref{eq:quadratic-field} into a matched
Gaussian polynomial.  A uniform stochastic approximation amounts to
coupling this polynomial to the empirical Taylor field while controlling
both the Taylor remainder and the replacement error at the scale of the
maximum.  Pointwise M-estimation theory supplies neither control.

When $a_S=0$, the quadratic field simplifies to
\begin{equation}\label{eq:orthogonal-quadratic-field}
 Q_S=-(\mathbb G_n\dot q_S^\star)^\top
 H_S^{-1}\mathbb G_nm_S^\star
 +\frac12P\ddot q_S^\star
 \left[H_S^{-1}\mathbb G_nm_S^\star,
       H_S^{-1}\mathbb G_nm_S^\star\right].
\end{equation}
The first term is a covariance-aware product of a target-derivative score
and a nuisance score.  The second records the curvature of the target map.
Orthogonality removes the remaining nuisance-curvature terms because they
are multiplied by $a_S$.

\section{Linear outcomes and logistic propensities}

Let $A\in\{0,1\}$, let $Y=AY(1)+(1-A)Y(0)$, and let $X\in\R^p$ satisfy
$\E X=0$ with covariance $C=\E(XX^\top)$.  The estimand is
$\theta_0=\E\{Y(1)-Y(0)\}$.  We begin with the additive linear outcome
model
\begin{equation}\label{eq:linear-potential-outcomes}
 Y(a)=\alpha_0+\tau_0a+X^\top b+\varepsilon_a,
 \qquad
 \E(\varepsilon_a\mid X)=0,
 \qquad a\in\{0,1\}.
\end{equation}
Thus $\theta_0=\tau_0$.  For $S\in\mathcal S_s$, put
$Z_S=(1,X_S^\top)^\top$ and
$W_S=(1,A,X_S^\top)^\top$.  The canonical outcome fit is the pooled
additive regression
\begin{equation}\label{eq:linear-outcome-models}
 \mu_S(a,X;\alpha_S)=\alpha_S^{(0)}+\tau_Sa+X_S^\top\beta_S,
 \qquad
 P_n[W_S\{Y-\mu_S(A,X;\hat\alpha_S)\}]=0.
\end{equation}
The propensity model on the same covariates is
\begin{equation}\label{eq:logistic-propensity}
 e_S(X;\gamma_S)=\Lambda(Z_S^\top\gamma_S),
 \qquad
 \Lambda(t)=\frac{e^t}{1+e^t},
\end{equation}
with empirical score
\begin{equation}\label{eq:logistic-score}
 P_n\{Z_S(A-e_S(X;\hat\gamma_S))\}=0.
\end{equation}
For AIPW, the restriction is joint: the single support $S$ is used for the
outcome slope and the propensity slope.  No sample splitting is imposed.

\section{IPW and AIPW}

The Horvitz--Thompson IPW estimator is
\begin{equation}\label{eq:ipw-map}
 \hat\theta_S^{\mathrm{IPW}}
 =P_n\left\{
 \frac{AY}{e_S(X;\hat\gamma_S)}
 -\frac{(1-A)Y}{1-e_S(X;\hat\gamma_S)}\right\}.
\end{equation}
Normalized weights can instead be included by stacking their two
normalizing equations with the logistic score.  If $e_S^\star$ denotes the
population logistic fit, then
\begin{align}
 J_S&=P\{e_S^\star(X)(1-e_S^\star(X))Z_SZ_S^\top\},
 \label{eq:fisher-information}\\
 a_S^{\mathrm{IPW}}
 &=-P\left[Z_S\left\{
       \frac{AY(1-e_S^\star)}{e_S^\star}
       +\frac{(1-A)Ye_S^\star}{1-e_S^\star}
       \right\}\right].
 \label{eq:ipw-derivative}
\end{align}
The nuisance Jacobian is $-J_S$.  Proposition~\ref{prop:first-order}
therefore adds
$(a_S^{\mathrm{IPW}})^\top J_S^{-1}
\mathbb G_n\{Z_S(A-e_S^\star)\}$ to the empirical IPW score.  This is the
linear channel through which propensity search responds to prognostic
covariates.

For candidate functions $e,\bar\mu_0,\bar\mu_1$, define
\begin{equation}\label{eq:aipw-map}
\begin{split}
 q^{\mathrm{AIPW}}(O;e,\bar\mu_0,\bar\mu_1)
 ={}&\bar\mu_1(X)-\bar\mu_0(X)
 +\frac{A}{e(X)}\{Y-\bar\mu_1(X)\}\\
 &-\frac{1-A}{1-e(X)}\{Y-\bar\mu_0(X)\}.
\end{split}
\end{equation}
The joint-support estimator is
\begin{equation}\label{eq:in-sample-aipw}
 \hat\theta_S^{\mathrm{AIPW}}
 =P_nq^{\mathrm{AIPW}}
 \{O;e_S(\,\cdot\,;\hat\gamma_S),
       \mu_S(0,\,\cdot\,;\hat\alpha_S),
       \mu_S(1,\,\cdot\,;\hat\alpha_S)\}.
\end{equation}

The population AIPW map has an exact mixed remainder.  Write
$e_0(X)=\Prob(A=1\mid X)$ and
$\mu_a(X)=\E\{Y(a)\mid X\}$.

\begin{proposition}[Exact mixed remainder]
\label{prop:aipw-mixed-remainder}
Suppose treatment is unconfounded given $X$.  If $\bar e$ is bounded away
from zero and one, then
\begin{equation}\label{eq:aipw-bias-identity}
\begin{split}
 &Pq^{\mathrm{AIPW}}(O;\bar e,\bar\mu_0,\bar\mu_1)-\theta_0\\
 &\quad=\E\left[(\bar e(X)-e_0(X))
 \left\{
 \frac{\bar\mu_1(X)-\mu_1(X)}{\bar e(X)}
 +\frac{\bar\mu_0(X)-\mu_0(X)}{1-\bar e(X)}
 \right\}\right].
\end{split}
\end{equation}
Thus the population target is correct if either nuisance component is
correct.  At $(e_0,\mu_0,\mu_1)$ every population nuisance derivative
vanishes.
\end{proposition}

\begin{proof}
Take the conditional expectation of \eqref{eq:aipw-map} given $X$ and
collect the two outcome-regression errors.  This gives
\eqref{eq:aipw-bias-identity}.  The right side is mixed at the truth, which
proves the remaining claims.
\end{proof}

Stacking \eqref{eq:linear-outcome-models} and
\eqref{eq:logistic-score} places \eqref{eq:in-sample-aipw} directly in the
generic expansion.  When every fitted nuisance is correct, the linear term
is common across supports and searched contrasts begin with the quadratic
field \eqref{eq:orthogonal-quadratic-field}.  More generally, the fitted
outcome regression itself changes the AIPW score.  The next result shows
that this change is exactly the leading search channel in the randomized
linear model.

\section{Coordinate-search bounds}

Suppose now that treatment is randomized independently of
$(X,Y(0),Y(1))$ with probability
$\pi\in[\epsilon,1-\epsilon]$.  Put
\begin{equation}\label{eq:outcome-signal}
 h_S=(Cb)_S,
\end{equation}
and define the leading search field
\begin{equation}\label{eq:common-leading-field}
 \ell_S=-\frac{1}{\pi(1-\pi)}
 h_S^\top C_{SS}^{-1}
 \frac1{\sqrt n}\sum_{i=1}^nX_{i,S}(A_i-\pi),
 \qquad \ell_\varnothing=0.
\end{equation}
Its signal scale is
\begin{equation}\label{eq:sparse-signal}
 \Psi_s=\max_{S\in\mathcal S_s}
 \{h_S^\top C_{SS}^{-1}h_S\}^{1/2}.
\end{equation}
The inverse is taken on the range of $C_{SS}$.  In particular,
$\Psi_s\le\norm{b}_C=(b^\top Cb)^{1/2}$.

Let
\begin{equation*}
 m_s=s\log(ep/s),
 \qquad
 \delta_s=\sqrt{m_s/n},
 \qquad
 \sigma_Y=\max_{a\in\{0,1\}}
 \norm{Y(a)}_{\psi_2}.
\end{equation*}
For the nonlinear fits, we use the localization quantities in
Proposition~\ref{prop:first-order}.  The regular search event is the event
on which all empirical roots exist and, for the IPW and joint AIPW systems
separately,
\begin{equation}\label{eq:regular-search-event}
 r_n\le K_0\delta_s,
 \qquad
 \epsilon_{m,n}\le K_0\delta_s,
 \qquad
 \epsilon_{q,n}\le K_0\sigma_Y\delta_s,
 \qquad
 \max_{S\in\mathcal S_s}\norm{a_S}\le K_0\sigma_Y.
\end{equation}
Sparse conditioning and relative moment concentration give the score and
Jacobian parts of this event.  Uniform control of the inverse propensity
weights supplies the remaining target-derivative condition.

\begin{proposition}[A common coordinate-search field]
\label{prop:coordinate-upper-bound}
Suppose \eqref{eq:linear-potential-outcomes} holds, all $2s$-sparse
eigenvalues of $C$ lie in $[\kappa^{-1},\kappa]$, and the centered augmented
row
\begin{equation*}
 V=\bigl(Y(0)-\E Y(0),Y(1)-\E Y(1),X^\top\bigr)^\top
\end{equation*}
obeys $\norm{u^\top V}_{\psi_2}\le K(u^\top\E VV^\top u)^{1/2}$ for every
$u$.  Suppose the regular search event \eqref{eq:regular-search-event} has
probability at least $1-C_0e^{-c_0m_s}$ for fixed $c_0,C_0>0$.
There are constants $c,C>0$, depending only on
$(\epsilon,\kappa,K,K_0,c_0,C_0)$, such that if $m_s\le cn$, then, with
probability at least $1-Ce^{-cm_s}$,
\begin{align}
 \max_{S\in\mathcal S_s}
 \left|\hat\theta_S^{\mathrm{IPW}}
 -\hat\theta_\varnothing^{\mathrm{IPW}}
 -\frac{\ell_S}{\sqrt n}\right|
 &\le C\sigma_Y\delta_s^2,
 \label{eq:ipw-coordinate-expansion}\\
 \max_{S\in\mathcal S_s}
 \left|\hat\theta_S^{\mathrm{AIPW}}
 -\hat\theta_\varnothing^{\mathrm{AIPW}}
 -\frac{\ell_S}{\sqrt n}\right|
 &\le C\sigma_Y\delta_s^2.
 \label{eq:aipw-coordinate-expansion}
\end{align}
Consequently, for $r\in\{\mathrm{IPW},\mathrm{AIPW}\}$,
\begin{equation}\label{eq:coordinate-upper-bound}
 \max_{S\in\mathcal S_s}
 |\hat\theta_S^r-\hat\theta_\varnothing^r|
 \le C\left\{
 \frac{\Psi_s}{\sqrt{\pi(1-\pi)}}\delta_s
 +\sigma_Y\delta_s^2\right\}.
\end{equation}
\end{proposition}

\begin{proof}
Random assignment makes the population propensity equal to $\pi$ for
every support.  The population slope in the additive outcome regression is
\begin{equation*}
 \beta_S=C_{SS}^{-1}(Cb)_S.
\end{equation*}
For IPW, the empirical score in \eqref{eq:ipw-map} is common across
supports at the population propensity.  After subtracting the
intercept-only fit, its nuisance-estimation correction is
\begin{equation*}
 -\frac{1}{\pi(1-\pi)}h_S^\top C_{SS}^{-1}
 \mathbb G_n\{X_S(A-\pi)\}=\ell_S.
\end{equation*}

For AIPW, the population outcome fits obey their projection equations.
Hence the population derivative with respect to the propensity coefficient
is zero on $Z_S$, while the derivatives with respect to the outcome
coefficients vanish because the propensity is correct.  Direct subtraction
of the two population AIPW maps gives
\begin{equation}\label{eq:aipw-score-difference}
 q_S^{\mathrm{AIPW},\star}-q_\varnothing^{\mathrm{AIPW},\star}
 =-\frac{A-\pi}{\pi(1-\pi)}
 X_S^\top C_{SS}^{-1}h_S.
\end{equation}
Thus IPW and joint-support AIPW have the same linear field $\ell_S$.

On the regular search event, Proposition~\ref{prop:first-order} bounds both
Taylor remainders by $C\sigma_Y\delta_s^2$, proving
\eqref{eq:ipw-coordinate-expansion} and
\eqref{eq:aipw-coordinate-expansion}.  Finally, a union bound over at most
$(ep/s)^s$ supports applied to \eqref{eq:common-leading-field} gives
\begin{equation*}
 \max_{S\in\mathcal S_s}|\ell_S|
 \le C\frac{\Psi_s}{\sqrt{\pi(1-\pi)}}\sqrt{m_s},
\end{equation*}
which proves \eqref{eq:coordinate-upper-bound}.
\end{proof}

The two terms in \eqref{eq:coordinate-upper-bound} separate the regimes.
Prognostic signal produces the first-order scale
$\Psi_s\sqrt{m_s/n}$; when $b=0$, searched contrasts begin at the
quadratic scale $m_s/n$.  Joint AIPW fitting does not remove the first term:
the support changes the outcome component of its score by exactly
\eqref{eq:aipw-score-difference}.

\section{Quadratic packing over irrelevant covariates}

The preceding bound does not distinguish the quadratic terms of outcome
regression, IPW, and AIPW.  A residual experiment makes that distinction
explicit.  Suppose every outcome model contains a fixed-dimensional,
correctly specified base and write the residual potential outcomes as
\begin{equation}\label{eq:residual-outcomes}
 Y(a)=\mu_a+M+\varepsilon_a,
 \qquad
 \E(M)=0,
 \qquad
 \E(\varepsilon_a\mid M)=0.
\end{equation}
The base spans $M$.  The search adds
$Z\sim N(0,I_{p_0})$, independent of
$(A,M,\varepsilon_0,\varepsilon_1)$.  Treatment is randomized with known
probability $\pi$, and the searched propensity is
\begin{equation*}
 e_S(Z)=\Lambda\{\operatorname{logit}(\pi)+Z_S^\top\gamma_S\}.
\end{equation*}
Let $\hat\theta_S^{\mathrm{OR}}$ be the treatment coefficient from pooled
least squares on the base, $A$, and $Z_S$.  In addition to the AIPW
estimator, consider the H\'ajek estimator
\begin{equation}\label{eq:hajek-ipw}
 \hat\theta_S^{\mathrm{IPW,H}}
 =\frac{P_n\{AY/e_S(Z;\hat\gamma_S)\}}
        {P_n\{A/e_S(Z;\hat\gamma_S)\}}
 -\frac{P_n[\{1-A\}Y/\{1-e_S(Z;\hat\gamma_S)\}]}
        {P_n[\{1-A\}/\{1-e_S(Z;\hat\gamma_S)\}]}.
\end{equation}
Normalization removes outcome-level terms that are unrelated to covariate
packing.

Put $w=\pi(1-\pi)$,
$\varepsilon=A\varepsilon_1+(1-A)\varepsilon_0$, and
\begin{equation}\label{eq:residual-packing-scores}
\begin{split}
 U_S&=\mathbb G_n\{Z_S(A-\pi)\},\\
 V_S^{\mathrm{OR}}&=\mathbb G_n(Z_S\varepsilon),\\
 V_S^{\mathrm{AIPW}}&=\mathbb G_n\{Z_S\varepsilon\omega_\pi(A)\},\\
 V_S^{\mathrm{IPW,H}}&=\mathbb G_n\{Z_S(M+\varepsilon)\omega_\pi(A)\},
\end{split}
\qquad
 \omega_\pi(A)=A\frac{1-\pi}{\pi}
 +(1-A)\frac{\pi}{1-\pi}.
\end{equation}

\begin{proposition}[Quadratic residual packing]
\label{prop:residual-packing}
Let $1\le q\le p_0/4$,
$m_0=q\log(ep_0/q)\to\infty$, and $m_0=o(n)$.  Suppose $M$ and the errors
in \eqref{eq:residual-outcomes} are sub-Gaussian, the error variances are
nonzero, and the conditions of Proposition~\ref{prop:second-order} hold for
the three fitted systems with $c_n=O(\sqrt{m_0})$.  Then, uniformly over
these supports,
\begin{equation}\label{eq:three-product-fields}
 \hat\theta_S^r-\hat\theta_\varnothing^r
 =-\frac{U_S^\top V_S^r}{nw}
 +o_{\Prob}(m_0/n),
 \qquad
 r\in\{\mathrm{OR},\mathrm{AIPW},\mathrm{IPW,H}\}.
\end{equation}
Define
\begin{align*}
 v_{\mathrm{OR}}
 &=\pi\Var(\varepsilon_1)+(1-\pi)\Var(\varepsilon_0),\\
 v_{\mathrm{AIPW}}
 &=\frac{(1-\pi)^2}{\pi}\Var(\varepsilon_1)
   +\frac{\pi^2}{1-\pi}\Var(\varepsilon_0),\\
 v_{\mathrm{IPW,H}}
 &=\frac{(1-\pi)^2}{\pi}\Var(M+\varepsilon_1)
   +\frac{\pi^2}{1-\pi}\Var(M+\varepsilon_0),
 \qquad
 \kappa_r=\sqrt{v_r/w}.
\end{align*}
Then
\begin{equation}\label{eq:residual-packing-rate}
 \max_{|S|\le q}
 (\hat\theta_S^r-\hat\theta_\varnothing^r)
 \asymp_{\Prob}\kappa_r\frac{m_0}{n}.
\end{equation}
Here $\asymp_{\Prob}$ means that the ratio is bounded above and bounded
away from zero in probability.
If the two errors have variance $\sigma_\varepsilon^2$, then
$\kappa_{\mathrm{OR}}\le\kappa_{\mathrm{AIPW}}$.  If also
$\Var(M+\varepsilon_a)=\sigma_M^2+\sigma_\varepsilon^2$, then
$\kappa_{\mathrm{AIPW}}\le\kappa_{\mathrm{IPW,H}}$.  At balance,
\begin{equation}\label{eq:balanced-packing-constants}
 \kappa_{\mathrm{OR}}=\kappa_{\mathrm{AIPW}}
 =2\sigma_\varepsilon,
 \qquad
 \kappa_{\mathrm{IPW,H}}
 =2\sqrt{\sigma_M^2+\sigma_\varepsilon^2}.
\end{equation}
When $\pi=1/2$, the OR and AIPW fields in
\eqref{eq:three-product-fields} are identical.
\end{proposition}

\begin{proof}
The logistic score gives
\begin{equation*}
 \hat\gamma_S=\frac{U_S}{w\sqrt n}+o_{\Prob}(\sqrt{m_0/n})
\end{equation*}
uniformly in $S$.  Frisch--Waugh--Lovell applied after the fixed base gives
the OR contrast $-U_S^\top V_S^{\mathrm{OR}}/(nw)$.

For AIPW, the empirical derivative with respect to the propensity slope at
the truth is $-V_S^{\mathrm{AIPW}}/\sqrt n$.  The empirical derivative with
respect to the common outcome slope contributes
$-U_S^\top V_S^{\mathrm{OR}}/(nw)$.  The population cross-derivative between
the two slopes contributes the opposite quantity.  These two terms cancel,
leaving the AIPW field in \eqref{eq:three-product-fields}.  Differentiating
the two normalized ratios in \eqref{eq:hajek-ipw} centers the outcome in
each arm and gives the IPW,H field.  The assumed Taylor remainder completes
\eqref{eq:three-product-fields}.

Conditional on $(A_i,M_i,\varepsilon_{0i},\varepsilon_{1i})_{i=1}^n$, the
coordinate pairs $(U_j,V_j^r)$ are independent Gaussian pairs across $j$.
Their variances converge to $(w,v_r)$ and their correlations converge to
zero.  Hence $-U_jV_j^r/w$ has two-sided exponential tails at scale
$\kappa_r$, uniformly on events whose probability tends to one.  The sum of
the largest $q$ positive values among $p_0$ such products is of order
$\kappa_rq\log(ep_0/q)$.  The upper bound follows by
truncating each product at a constant multiple of
$\kappa_r\log(ep_0/q)$ and summing its exponential tail.  For the lower
bound, a smaller multiple of this threshold is exceeded by at least $q$
coordinates with probability tending to one, by a binomial concentration
bound.  This proves \eqref{eq:residual-packing-rate}.  The variance formulas
follow from \eqref{eq:residual-packing-scores}.  Under homoskedasticity,
\begin{equation*}
 \frac{v_{\mathrm{AIPW}}}{\sigma_\varepsilon^2}
 =\frac{(1-\pi)^2}{\pi}+\frac{\pi^2}{1-\pi}\ge1,
\end{equation*}
with equality only at balance.  Substitution gives the remaining
comparisons and \eqref{eq:balanced-packing-constants}.
\end{proof}

At balance, AIPW reduces the IPW packing constant by
\begin{equation*}
 \frac{\sigma_\varepsilon}
 {\sqrt{\sigma_M^2+\sigma_\varepsilon^2}}=\sqrt{1-R_M^2},
 \qquad
 R_M^2=\frac{\sigma_M^2}
 {\sigma_M^2+\sigma_\varepsilon^2}.
\end{equation*}
This is the precise residualization advantage.  It does not change the
$m_0/n$ rate.  Under unequal treatment probabilities, AIPW pays the factor
\begin{equation*}
 \left\{\frac{(1-\pi)^2}{\pi}
 +\frac{\pi^2}{1-\pi}\right\}^{1/2}
\end{equation*}
relative to OR under homoskedasticity.

\section{Planted confounding and deliberate omission}

The quadratic comparison leaves open how the researcher chooses among
population-misspecified supports.  Consider
\begin{equation}\label{eq:planted-confounding-model}
 X\sim N(0,C),
 \qquad
 e_0(X)=\Lambda(\alpha+X^\top\gamma),
 \qquad
 Y(a)=\mu+\tau a+X^\top\beta+\varepsilon_a,
\end{equation}
where $C$ is positive definite, treatment is unconfounded given $X$, and
$\E(\varepsilon_a\mid X)=0$.  Put
$K=\operatorname{supp}(\gamma)\cup\operatorname{supp}(\beta)$.  For a
support $T$, let $U=K\setminus T$ and define
\begin{equation*}
 C_{UU\cdot T}=C_{UU}-C_{UT}C_{TT}^{-1}C_{TU}.
\end{equation*}
Let $\theta_T^{\mathrm{OR}}$ be the population coefficient on $A$ in the
pooled regression on $(1,A,X_T)$.  Let $e_T^\star$ be the population
logistic fit on $(1,X_T)$, and let
$\bar\mu_{a,T}^\star$ be the population pooled additive outcome fit on
$(1,A,X_T)$.

\begin{proposition}[Omitted-variable objective]
\label{prop:omitted-variable-objective}
Put $\pi=\E e_0(X)$, $w=\pi(1-\pi)$, and
$c_\gamma=\E\Lambda'(\alpha+X^\top\gamma)$.  If the residual variance of
$A$ after its linear projection on $X_T$ is positive, then
\begin{equation}\label{eq:exact-or-ovb}
 \theta_T^{\mathrm{OR}}-\tau
 =\frac{c_\gamma\,
 \gamma_U^\top C_{UU\cdot T}\beta_U}
 {w-c_\gamma^2(C\gamma)_T^\top
 C_{TT}^{-1}(C\gamma)_T}.
\end{equation}
Consequently, omission raises the population OR estimand above $\tau$ if
and only if
\begin{equation}\label{eq:ovb-alignment}
 \gamma_U^\top C_{UU\cdot T}\beta_U>0.
\end{equation}

The corresponding population IPW bias obeys the exact identity
\begin{equation}
 \theta_T^{\mathrm{IPW}}-\tau
 =\E\left[(e_0(X)-e_T^\star(X))\left\{
 \frac{\mu_1(X)}{e_T^\star(X)}
 +\frac{\mu_0(X)}{1-e_T^\star(X)}
 \right\}\right],
 \label{eq:exact-ipw-ovb}
\end{equation}
Here the label $\theta_T^{\mathrm{IPW}}$ denotes the Horvitz--Thompson
population functional.  The exact AIPW bias is
\eqref{eq:aipw-bias-identity} with
$(\bar e,\bar\mu_0,\bar\mu_1)
=(e_T^\star,\bar\mu_{0,T}^\star,\bar\mu_{1,T}^\star)$.

For the local propensity model
$e_{0,t}(X)=\Lambda\{\alpha_t+tX^\top\bar\gamma\}$, choose $\alpha_t$ so
that $\E e_{0,t}(X)=\pi$.  For every fixed $T$, as $t\to0$,
\begin{equation}\label{eq:common-local-ovb}
 \theta_T^r-\tau
 =t\,\bar\gamma_U^\top C_{UU\cdot T}\beta_U+O(t^2),
 \qquad
 r\in\{\mathrm{OR},\mathrm{IPW},\mathrm{AIPW}\}.
\end{equation}
Thus the three estimators have the same first local omitted-variable
objective when the propensity and outcome models use the same support.
\end{proposition}

\begin{proof}
Gaussian integration by parts gives
\begin{equation*}
 d:=\operatorname{Cov}(X,A)=\E\{Xe_0(X)\}=c_\gamma C\gamma.
\end{equation*}
Frisch--Waugh--Lovell therefore gives
\begin{equation*}
 \theta_T^{\mathrm{OR}}-\tau
 =\frac{(d_U-C_{UT}C_{TT}^{-1}d_T)^\top\beta_U}
 {w-d_T^\top C_{TT}^{-1}d_T},
\end{equation*}
which is \eqref{eq:exact-or-ovb}.  Its denominator is positive and
$c_\gamma>0$, proving \eqref{eq:ovb-alignment}.  Conditioning the IPW
functional on $X$ gives \eqref{eq:exact-ipw-ovb}; the AIPW identity is
Proposition~\ref{prop:aipw-mixed-remainder}.

Write
\begin{equation*}
 R_{U\mid T}=X_U-C_{UT}C_{TT}^{-1}X_T.
\end{equation*}
The population logistic score and a first-order expansion at $t=0$ give
\begin{equation*}
 e_T^\star-e_{0,t}
 =-wt\,\bar\gamma_U^\top R_{U\mid T}+O_{L_2}(t^2).
\end{equation*}
At $t=0$, the pooled outcome projection gives, for both arms,
\begin{equation*}
 \bar\mu_{a,T}^\star-\mu_a
 =-\beta_U^\top R_{U\mid T}+O_{L_2}(t).
\end{equation*}
Substitution in \eqref{eq:exact-ipw-ovb} and
\eqref{eq:aipw-bias-identity} gives
$t\E\{(\bar\gamma_U^\top R_{U\mid T})
(\beta_U^\top R_{U\mid T})\}+O(t^2)$, which is
\eqref{eq:common-local-ovb}.  The OR expansion follows directly from
\eqref{eq:exact-or-ovb}.
\end{proof}

Under $C=I$, put $c_j=\bar\gamma_j\beta_j$.  The local upward-bias
objective becomes
\begin{equation}\label{eq:independent-omission-objective}
 t\sum_{j\notin T}c_j.
\end{equation}
With a budget $|T|\le s$, a population hacker therefore includes up to
$s$ active variables with the most negative $c_j$, omits every variable
with $c_j>0$, and is indifferent among null variables.  If the planted
support has size $k\le s$, correct specification is feasible but need not
maximize the estimand: any positive $c_j$ is deliberately omitted.  If
$k>s$, the budget is spent on the most negative signed products, not the
largest coefficients in absolute value.  In particular, if every $c_j$ is
positive and enough null variables are available, the maximizing support
contains no planted variable.

Away from the local regime, even the OR denominator affects the support
choice.  Under $C=I$, adding $j\notin T$ raises its population estimand
exactly when
\begin{equation}\label{eq:exact-add-variable-rule}
 c_\gamma^2\gamma_j^2
 \sum_{\ell\notin T}\gamma_\ell\beta_\ell
 >\{w-c_\gamma^2\norm{\gamma_T}^2\}\gamma_j\beta_j.
\end{equation}
For $\gamma=t\bar\gamma$, this reduces to $c_j<0$ at first order.  There is
no corresponding global ordering of OR, IPW, and AIPW without further
restrictions: \eqref{eq:exact-ipw-ovb} and
\eqref{eq:aipw-bias-identity} weight the same misspecification differently.

The alignment criterion also connects directly to sparse isotropy.  If
$J=T\cup U$ satisfies
$\norm{C_{JJ}-I}_{\mathrm{op}}\le\rho<1$, then
\begin{equation}\label{eq:rip-ovb-stability}
 \left|\gamma_U^\top(C_{UU\cdot T}-I)\beta_U\right|
 \le\frac{\rho}{1-\rho}\norm{\gamma_U}\norm{\beta_U}.
\end{equation}
Thus the independent-coordinate sign rule is stable whenever its signed
margin exceeds the right side.  More generally, the Schur complement in
\eqref{eq:ovb-alignment} shows that correlated proxies act as groups:
including $X_T$ removes exactly the component of each omitted direction
linearly explained by $X_T$.

Propositions~\ref{prop:residual-packing} and
\ref{prop:omitted-variable-objective} describe different regimes.  Within
a fixed active model, irrelevant covariates generate quadratic packing and
residualization orders the constants.  Across active models, deliberate
omission creates a population objective.  In the weak-confounding limit
that objective is common to OR, IPW, and AIPW, so AIPW's need for aligned
nuisance errors does not create an additional combinatorial barrier.  Its
advantage appears after the shared confounding signal has been removed:
the outcome residual replaces the raw outcome in the product field.
For a fixed collection of active supports, an omitted-variable gap larger
than $n^{-1/2}$ determines the selected population model asymptotically;
for a growing collection, $n^{-1/2}$ is replaced by the corresponding
uniform linear-field scale.  Conditional on that active model, irrelevant
fillers contribute the smaller quadratic scale in
\eqref{eq:residual-packing-rate}.  These statements concern maximization of
the point estimate.  Searching a studentized statistic also changes its
estimated denominator and requires a separate expansion.

\section{Extended search over covariate subspaces}

The support search has a basis-free extension.  Let $\mathfrak V_s$ be the
collection of subspaces $V\subseteq\R^p$ with $\dim(V)\le s$.  In model
$V$, each covariate slope is restricted to $V$.  For AIPW the same $V$ is
imposed jointly on the propensity and additive outcome regressions.
Write $\hat\theta_V^r$ for the resulting estimator and
$\hat\theta_0^r$ for its intercept-only value.

\begin{proposition}[Exact rank reduction]
\label{prop:rank-reduction}
Fix a sample on which every fitted model below has a finite, unique fitted
linear predictor.  For every $s\ge1$,
\begin{align}
 \sup_{V\in\mathfrak V_s}\hat\theta_V^{\mathrm{IPW}}
 &=\sup_{V\in\mathfrak V_1}\hat\theta_V^{\mathrm{IPW}},
 \label{eq:ipw-rank-reduction}\\
 \sup_{V\in\mathfrak V_s}\hat\theta_V^{\mathrm{AIPW}}
 &=\sup_{V\in\mathfrak V_{\min(s,2)}}
   \hat\theta_V^{\mathrm{AIPW}}.
 \label{eq:aipw-rank-reduction}
\end{align}
\end{proposition}

\begin{proof}
Given $V$, the fitted logistic slope spans a subspace of dimension at most
one.  Restricting the logistic problem to this span preserves its fitted
predictor and its score equations, proving \eqref{eq:ipw-rank-reduction}.
For AIPW, take the span of the fitted logistic and outcome slopes.  Its
dimension is at most two.  Both fitted predictors remain feasible, the
logistic score remains zero, and the OLS residual remains orthogonal to the
smaller space.  The AIPW value is
unchanged, which proves \eqref{eq:aipw-rank-reduction}; the reverse
inequalities follow by inclusion.
\end{proof}

The rank reduction is exact, but it does not by itself evaluate the
continuum search.  In the local regime the common field does.  Suppose now
that $C$ is positive definite and define
\begin{equation}\label{eq:whitened-field}
 g_n=\frac1{\sqrt n}\sum_{i=1}^nX_i(A_i-\pi),
 \qquad
 z_n=C^{-1/2}g_n.
\end{equation}

\begin{proposition}[Local extended-hacking bound]
\label{prop:extended-upper-bound}
Suppose the randomized linear model and relative sub-Gaussian condition of
Proposition~\ref{prop:coordinate-upper-bound} hold, the eigenvalues of $C$
are bounded away from zero and infinity, and $p=o(n)$.  Suppose the
unpenalized logistic fits are finite and the analogue of
\eqref{eq:regular-search-event} holds uniformly over
$V\in\mathfrak V_s$, with $\delta_s$ replaced by $\sqrt{p/n}$ and bounded
constants interpreted in probability.  Then, for either
$r\in\{\mathrm{IPW},\mathrm{AIPW}\}$ and every $s\ge1$,
\begin{equation}\label{eq:extended-expansion}
 \sup_{V\in\mathfrak V_s}
 (\hat\theta_V^r-\hat\theta_0^r)
 =\frac{\norm{b}_C\norm{z_n}-b^\top g_n}
 {2\pi(1-\pi)\sqrt n}
 +O_{\Prob}\left(\frac{\sigma_Yp}{n}\right).
\end{equation}
In particular,
\begin{equation}\label{eq:extended-upper-bound}
 \sup_{V\in\mathfrak V_s}
 (\hat\theta_V^r-\hat\theta_0^r)
 =O_{\Prob}\left\{
 \frac{\norm{b}_C}{\sqrt{\pi(1-\pi)}}\sqrt{\frac pn}
 +\frac{\sigma_Yp}{n}\right\}.
\end{equation}
\end{proposition}

\begin{proof}
For $V\subseteq\R^p$, let $P_V$ denote the Euclidean projector onto
$C^{1/2}V$.  The population outcome projection replaces $C^{1/2}b$ by
$P_VC^{1/2}b$.  The calculation in
Proposition~\ref{prop:coordinate-upper-bound} therefore gives the common
leading contrast
\begin{equation*}
 -\frac{(C^{1/2}b)^\top P_Vz_n}{\pi(1-\pi)\sqrt n}.
\end{equation*}
Uniform full-block localization gives a remainder
$O_{\Prob}(\sigma_Yp/n)$.  Maximizing the leading term over projectors of
rank at most $s$ reduces to the largest positive eigenvalue of
$-\{(C^{1/2}b)z_n^\top+z_n(C^{1/2}b)^\top\}/2$.  This matrix has at most
one positive eigenvalue, equal to
$(\norm{b}_C\norm{z_n}-b^\top g_n)/2$; a rank-one subspace attains it.
This proves \eqref{eq:extended-expansion}.  Since
$\norm{z_n}=O_{\Prob}(\sqrt{\pi(1-\pi)p})$, the upper bound follows.
\end{proof}

The extended result replaces the coordinate complexity
$s\log(ep/s)$ by $p$.  Its first-order maximizer is one-dimensional even
though exact joint AIPW fitting may require two directions.  Unlike
least squares, unpenalized logistic fitting can fail under separation, so
\eqref{eq:extended-expansion} is a local theorem rather than a global
proportional-limit formula.  A strictly convex, basis-invariant penalty on
the fitted linear predictor preserves Proposition~\ref{prop:rank-reduction}
and supplies a well-defined alternative beyond that event.

The fully interacted outcome fit, with a separate slope in each arm, is a
direct extension rather than the canonical construction.  Under
$Y(a)=\mu_a+X^\top b_a+\varepsilon_a$, the same bounds hold after replacing
$b$ by $(1-\pi)b_1+\pi b_0$; the exact AIPW rank bound becomes three.
