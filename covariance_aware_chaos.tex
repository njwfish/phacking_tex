\begin{abstract}
For a general control covariance, the largest coefficient obtained by
searching over $s$ controls reduces to a covariance-normalized bilinear
functional of two empirical score vectors.  Under relative
sub-Gaussianity and $[s\log(ep/s)]^4=o(n)$, this functional admits a
Gaussian approximation with the same score covariance, including its
fourth-moment dependence.  An oracle Gaussian multiplier gives the
corresponding conditional approximation from the row scores.
Conditional-product greedy maximizes the exact Schur-complement increments
and becomes top-$s$ sorting when the controls are uncorrelated.  An exchange
argument bounds its regret by the aggregate oscillation of each coordinate's
marginal across histories of the same depth, yielding a score-adaptive
stochastic condition verified when $\rho_s(C)\sqrt{s}\to0$.  For centered,
separable score covariances, a second argument uses coordinates uncorrelated
with the current support and permits sparse covariance graphs with fixed
local correlations.
\end{abstract}

\section{Gaussian score approximation}

For independent and identically distributed observations
$i=1,\ldots,n$, consider the centered linear model
\begin{equation}\label{eq:model}
        y^{(i)}
        =\beta_0x_0^{(i)}
        +(X_{-0}^{(i)})^\top\beta_{-0}
        +\varepsilon^{(i)}.
\end{equation}
The pair $(x_0^{(i)},X_{-0}^{(i)})$ is jointly sub-Gaussian relative to
its covariance, and $\varepsilon^{(i)}$ is centered, sub-Gaussian, and
independent of the design.  Write
\begin{equation*}
        \Var(x_0^{(i)})=\sigma_0^2,
        \qquad
        \Cov(X_{-0}^{(i)})=C,
        \qquad
        d=\Cov(X_{-0}^{(i)},x_0^{(i)}),
\end{equation*}
where $\sigma_0>0$.  Every standardized linear form of the design and
noise has uniformly bounded $\psi_2$ norm.  Rescaling a control leaves
every selected column span unchanged, so we assume throughout that
$C_{jj}=1$ for every $j$.

Write $[p]=\{1,\ldots,p\}$.  For $S\subseteq[p]$, let $M_S$ project
orthogonally off the column span of $X_{-0,S}$, and define
\begin{equation}\label{eq:searched-coefficient}
        \widehat\beta_{0,S}
        =\frac{x_0^\top M_Sy}{x_0^\top M_Sx_0}.
\end{equation}
The target is the supremum of \eqref{eq:searched-coefficient} over
$|S|=s$.  Define the residual outcome and its score mean by
\begin{equation}\label{eq:residual-outcome}
\begin{aligned}
        r^{(i)}
        &=y^{(i)}-\beta_0x_0^{(i)}
        -\frac{d^\top\beta_{-0}}{\sigma_0^2}x_0^{(i)},\\
        h&=\Cov(X_{-0}^{(i)},r^{(i)})
        =C\beta_{-0}
        -\frac{d^\top\beta_{-0}}{\sigma_0^2}d,
        &\sigma_r^2&=\Var(r^{(i)}).
\end{aligned}
\end{equation}
Then $\Cov(x_0^{(i)},r^{(i)})=0$.  We assume $\sigma_r>0$.

For $v\in\R^p$ and $1\le k\le2s$, define the
covariance-normalized sparse norm
\begin{equation}\label{eq:sparse-norm}
        \norm{v}_{C,k}
        =\sup_{J\subseteq[p],\,|J|\le k}
        \left(v_J^\top C_{JJ}^{-1}v_J\right)^{1/2},
\end{equation}
and assume that the blocks $C_{JJ}$ are positive definite for
$|J|\le2s$.  Write
\begin{equation}\label{eq:scales}
        L=\log(ep/s),
        \qquad
        \delta_{s,n,p}=\sqrt{sL/n}.
\end{equation}
The population score means are diffuse when
\begin{equation}\label{eq:proxy-energy}
        \norm{d}_{C,2s}
        \lesssim\sigma_0\delta_{s,n,p},
        \qquad
        \norm{h}_{C,2s}
        \lesssim\sigma_r\delta_{s,n,p}.
\end{equation}

For $a,b\in\R^p$ and $S\subseteq[p]$ with $|S|\le2s$, put
\begin{equation}\label{eq:search-functional}
        Q_S(a,b)=-a_S^\top C_{SS}^{-1}b_S,
        \qquad
        T_C(a,b)=\max_{|S|=s}Q_S(a,b).
\end{equation}
Define the centered row score and its covariance by
\begin{equation}\label{eq:matched-score}
        U^{(i)}=
        \begin{pmatrix}
        x_0^{(i)}X_{-0}^{(i)}-d\\
        r^{(i)}X_{-0}^{(i)}-h
        \end{pmatrix},
        \qquad
        \Sigma_U=\Var(U^{(i)}),
\end{equation}
and collect the empirical scores in
\begin{equation}\label{eq:score-sum}
        Z_n=\begin{pmatrix}Z_{1n}\\Z_{2n}\end{pmatrix}
        :=\begin{pmatrix}
        X_{-0}^\top x_0/\sqrt n\\
        X_{-0}^\top r/\sqrt n
        \end{pmatrix}
        =\begin{pmatrix}\sqrt n\,d\\\sqrt n\,h\end{pmatrix}
        +\frac1{\sqrt n}\sum_{i=1}^nU^{(i)}.
\end{equation}
The matched Gaussian score vector is
\begin{equation}\label{eq:matched-gaussian}
        Z=\begin{pmatrix}Z_1\\Z_2\end{pmatrix}
        \sim N\!\left(
        \begin{pmatrix}\sqrt n\,d\\\sqrt n\,h\end{pmatrix},
        \Sigma_U\right).
\end{equation}
The covariance $\Sigma_U$ contains the fourth-moment contribution of the
design.

For a pair $z=(z_1,z_2)$, write
\begin{equation}\label{eq:standardized-sparse-norm}
        |z|_s
        =\frac{\norm{z_1}_{C,s}}{\sigma_0}
        +\frac{\norm{z_2}_{C,s}}{\sigma_r}.
\end{equation}

For real random variables $\zeta,\widetilde\zeta$, write
\begin{equation*}
        d_3(\zeta,\widetilde\zeta)
        =\sup_{\varphi}
        |\E\varphi(\zeta)-\E\varphi(\widetilde\zeta)|.
\end{equation*}
The supremum is over all three times continuously differentiable
$\varphi$ such that
$\max_{0\le j\le3}\norm{\varphi^{(j)}}_\infty\le1$.

\begin{lemma}[Sparse score moments]
\label{lem:sparse-score-moments}
Let $\widetilde U^{(1)},\ldots,\widetilde U^{(n)}$ be independent, with
each $\widetilde U^{(i)}$ distributed either as $U^{(i)}$ or as a
centered Gaussian vector with covariance $\Sigma_U$.  Suppose
$sL=o(n)$.  Uniformly over all such choices and every integer
$1\le q\le6$,
\begin{equation}\label{eq:sparse-score-moments}
\begin{split}
        \E|\widetilde U^{(i)}|_s^q&\lesssim(sL)^{q/2},\\
        \E\left|
        \begin{pmatrix}\sqrt n\,d\\\sqrt n\,h\end{pmatrix}
        +\frac1{\sqrt n}\sum_{i=1}^n\widetilde U^{(i)}
        \right|_s^q
        &\lesssim(sL)^{q/2}.
\end{split}
\end{equation}
\end{lemma}

\begin{proof}
For one row, the product structure in \eqref{eq:matched-score} gives
\begin{equation}\label{eq:row-score-factorization}
\begin{split}
        |U^{(i)}|_s
        \lesssim{}&
        \left(
        \frac{|x_0^{(i)}|}{\sigma_0}
        +\frac{|r^{(i)}|}{\sigma_r}
        \right)
        \sup_{|J|\le s}
        \left\{(X_{-0,J}^{(i)})^\top
        C_{JJ}^{-1}X_{-0,J}^{(i)}\right\}^{1/2}\\
        &+\frac{\norm{d}_{C,s}}{\sigma_0}
        +\frac{\norm{h}_{C,s}}{\sigma_r}.
\end{split}
\end{equation}
For each $J$, a fixed net of the $C_{JJ}$-unit sphere has at most
$9^{|J|}$ points, and there are at most $(ep/s)^s$ supports.  The
relative sub-Gaussian assumption and integration of the resulting union
bound show that the supremum in \eqref{eq:row-score-factorization} has
$L_q$ norm $O(\sqrt{sL})$ for $q\le12$.  The two scalar factors have
bounded normalized $L_q$ norms, so H\"older's inequality and
\eqref{eq:proxy-energy} prove the first claim for $U^{(i)}$.  The
covariance of every normalized projection is uniformly bounded, so the
same support-net argument proves it for the matched Gaussian choice.

For the second claim, fix a net direction and let $V_i$ be its centered
scalar projection of $\widetilde U^{(i)}$.  This projection is either
Gaussian or a centered sum of products of sub-Gaussian variables, so
Bernstein's inequality gives
\begin{equation*}
        \Prob\left(
        \left|\frac1{\sqrt n}\sum_{i=1}^nV_i\right|>t
        \right)
        \le2\exp\{-c\min(t^2,t\sqrt n)\}
\end{equation*}
for each net direction, with a uniform constant.  A union over the same
supports and nets, followed by tail integration, gives an
$O(\sqrt{sL})$ bound in $L_q$ because $sL=o(n)$.  The proxy-energy
condition \eqref{eq:proxy-energy} makes the standardized sparse norm of
the deterministic mean $O(\sqrt{sL})$.
\end{proof}

\begin{theorem}[Gaussian approximation]
\label{thm:matched-chaos}
Suppose the model above satisfies \eqref{eq:proxy-energy} and
$sL=o(n)$.  Then
\begin{equation}\label{eq:empirical-chaos}
\begin{split}
        \sup_{S\subseteq[p],\,|S|=s}\widehat\beta_{0,S}
        &=\widehat\beta_{0,\emptyset}
        +\frac{T_C(Z_{1n},Z_{2n})}{n\sigma_0^2}
        +O_{\Prob}\!\left(
        \frac{\sigma_r}{\sigma_0}\delta_{s,n,p}^3
        \right).
\end{split}
\end{equation}

If in addition $(sL)^4=o(n)$, then the matched Gaussian vector in
\eqref{eq:matched-gaussian} satisfies
\begin{equation}\label{eq:smooth-approximation}
        d_3\!\left(
        \frac{n\sigma_0}{\sigma_rsL}
        \left\{
        \sup_{S\subseteq[p],\,|S|=s}\widehat\beta_{0,S}
        -\widehat\beta_{0,\emptyset}
        \right\},
        \frac{T_C(Z_1,Z_2)}{\sigma_0\sigma_rsL}
        \right)
        \lesssim
        \left(\frac{(sL)^4}{n}\right)^{1/6}
        +o(1).
\end{equation}
In particular, the two scalar laws can be coupled so that
\begin{equation}\label{eq:coupled-approximation}
        \sup_{S\subseteq[p],\,|S|=s}\widehat\beta_{0,S}
        =\widehat\beta_{0,\emptyset}
        +\frac{T_C(Z_1,Z_2)}{n\sigma_0^2}
        +o_{\Prob}\!\left(
        \frac{\sigma_r}{\sigma_0}\frac{sL}{n}
        \right).
\end{equation}
\end{theorem}

\begin{proof}
The empirical reduction follows from an exact regression identity.  With
$\widehat C=X_{-0}^\top X_{-0}/n$, block projection gives
\begin{equation}\label{eq:exact-regression}
        \widehat\beta_{0,S}-\widehat\beta_{0,\emptyset}
        =
        \frac{-Z_{1n,S}^\top\widehat C_{SS}^{-1}Z_{2n,S}
        +\dfrac{x_0^\top r}{x_0^\top x_0}
        Z_{1n,S}^\top\widehat C_{SS}^{-1}Z_{1n,S}}
        {x_0^\top x_0-Z_{1n,S}^\top
        \widehat C_{SS}^{-1}Z_{1n,S}}.
\end{equation}
The relative sub-Gaussian assumption and a sparse-net argument yield the
covariance-normalized block-Gram event
\begin{equation}\label{eq:block-gram}
        \sup_{|J|\le2s}
        \norm{C_{JJ}^{-1/2}\widehat C_{JJ}C_{JJ}^{-1/2}-I}_{\mathrm{op}}
        =O_{\Prob}(\delta_{s,n,p}).
\end{equation}
The same argument, now applied to the product scores, gives
\begin{equation}\label{eq:score-size}
        \norm{Z_{1n}}_{C,s}=O_{\Prob}(\sigma_0\sqrt{sL}),
        \qquad
        \norm{Z_{2n}}_{C,s}=O_{\Prob}(\sigma_r\sqrt{sL}).
\end{equation}
The proxy bound \eqref{eq:proxy-energy} covers the means in
\eqref{eq:score-size}.  The remaining scalar terms satisfy
\begin{equation*}
        x_0^\top x_0
        =n\sigma_0^2\{1+O_{\Prob}(n^{-1/2})\},
        \qquad
        \frac{x_0^\top r}{x_0^\top x_0}
        =O_{\Prob}\!\left(\frac{\sigma_r}{\sigma_0\sqrt n}\right),
\end{equation*}
because $x_0^{(i)}$ and $r^{(i)}$ are uncorrelated sub-Gaussian
variables.  Substitution into \eqref{eq:exact-regression}, followed by
the inverse perturbation bound on \eqref{eq:block-gram}, proves
\eqref{eq:empirical-chaos}.

For the Gaussian approximation, write
$\widetilde Q_S(z)=Q_S(z_1,z_2)/(\sigma_0\sigma_r)$ and smooth the
supremum with $\lambda>0$ by
\begin{equation*}
        f_\lambda(z)
        =\frac1\lambda
        \log\sum_{|S|=s}\exp\{\lambda\widetilde Q_S(z)\}.
\end{equation*}
Since $\log\binom ps\le sL$, replacing
$T_C(z_1,z_2)/(\sigma_0\sigma_rsL)$ by $f_\lambda(z)/(sL)$ costs at most
$1/\lambda$.

Each normalized support value obeys, for directions $u,v$,
\begin{equation*}
        |D\widetilde Q_S(z)[u]|\le |z|_s|u|_s,
        \qquad
        |D^2\widetilde Q_S(z)[u,v]|\le |u|_s|v|_s.
\end{equation*}
Softmax differentiation therefore gives
\begin{equation}\label{eq:softmax-derivatives}
\begin{aligned}
        |Df_\lambda(z)[u]|&\lesssim |z|_s|u|_s,\\
        |D^2f_\lambda(z)[u,u]|&\lesssim
        \{1+\lambda|z|_s^2\}|u|_s^2,\\
        |D^3f_\lambda(z)[u,u,u]|&\lesssim
        \{\lambda|z|_s+\lambda^2|z|_s^3\}|u|_s^3.
\end{aligned}
\end{equation}
Let $\varphi$ belong to the class defining $d_3$ and apply Taylor's
formula to $H(z)=\varphi\{f_\lambda(z)/(sL)\}$ while replacing the $n$
score rows by matched Gaussian rows one at a time.  The chain rule and
\eqref{eq:softmax-derivatives} give
\begin{equation}\label{eq:composite-third-derivative}
\begin{split}
        |D^3H(z)[u,u,u]|
        \lesssim |u|_s^3\bigg\{
        &\frac{|z|_s^3}{(sL)^3}
        +\frac{|z|_s}{(sL)^2}
        +\frac{\lambda|z|_s^3}{(sL)^2}\\
        &+\frac{\lambda|z|_s}{sL}
        +\frac{\lambda^2|z|_s^3}{sL}
        \bigg\}.
\end{split}
\end{equation}
The first two Taylor terms cancel because the rows have the same mean and
covariance.  Lemma~\ref{lem:sparse-score-moments} and
\eqref{eq:composite-third-derivative} bound the sum of the third-order
remainders by
\begin{equation}\label{eq:lindeberg-rate}
        \frac{1+\lambda sL+\lambda^2(sL)^2}{\sqrt n}.
\end{equation}
Balancing \eqref{eq:lindeberg-rate} with the smoothing error using
$\lambda=\{\sqrt n/(sL)^2\}^{1/3}$ gives
$\{(sL)^4/n\}^{1/6}$.  This proves \eqref{eq:smooth-approximation} after
using \eqref{eq:empirical-chaos}.  The normalized statistics are tight
by \eqref{eq:score-size}; smoothing the scalar test functions and applying
the standard coupling characterization of weak convergence then yields
\eqref{eq:coupled-approximation}.
\end{proof}

A Gaussian multiplier estimates $\Sigma_U$ from the row scores.
Let $\bar U=n^{-1}\sum_iU^{(i)}$ and define
\begin{equation*}
        \widehat\Sigma_U
        =\frac1n\sum_{i=1}^n
        (U^{(i)}-\bar U)(U^{(i)}-\bar U)^\top.
\end{equation*}
Conditionally on the data, let $e_i\sim N(0,1)$ be independent and set
\begin{equation}\label{eq:multiplier}
        Z^\#=\begin{pmatrix}Z_1^\#\\Z_2^\#\end{pmatrix}
        :=\begin{pmatrix}\sqrt n\,d\\\sqrt n\,h\end{pmatrix}
        +\frac1{\sqrt n}\sum_{i=1}^ne_i(U^{(i)}-\bar U).
\end{equation}
Conditionally on the observations, $Z^\#$ is Gaussian with the same mean
as $Z$ and covariance $\widehat\Sigma_U$.  It is oracle because $d$, $h$,
and $r$ contain population quantities.

\begin{lemma}[Sparse multiplier covariance]
\label{lem:multiplier-covariance}
For $J\subseteq[p]$, let $U^{(i)}[J]$, $\Sigma_U[J]$, and
$\widehat\Sigma_U[J]$ contain both score channels indexed by $J$, and put
\begin{equation*}
        \mathcal W_J
        =\diag(\sigma_0^{-1}C_{JJ}^{-1/2},
        \sigma_r^{-1}C_{JJ}^{-1/2}).
\end{equation*}
Under the assumptions of Theorem~\ref{thm:matched-chaos},
\begin{equation}\label{eq:multiplier-covariance-rate}
\begin{split}
        \eta_n
        &:=
        \sup_{|J|\le2s}
        \norm{\mathcal W_J
        \{\widehat\Sigma_U[J]-\Sigma_U[J]\}
        \mathcal W_J^\top}_{\mathrm{op}}\\
        &=O_{\Prob}\!\left(
        \sqrt{\frac{sL}{n}}+\frac{(sL)^2}{n}
        \right).
\end{split}
\end{equation}
\end{lemma}

\begin{proof}
Every linear form of $\mathcal W_JU^{(i)}[J]$ is a sum of centered
products of normalized sub-Gaussian variables and therefore has uniformly
bounded $\psi_1$ norm.  Its centered square has bounded
$\psi_{1/2}$ norm.  A Fuk--Nagaev truncation bound for the average of
these squares, followed by a $1/4$-net on each sphere and a union over the
at most $\exp(c_1sL)$ support-net pairs, gives
\begin{equation*}
        \sqrt{\frac{sL+t}{n}}
        +\frac{(sL+t)^2}{n}
\end{equation*}
with probability at least $1-2e^{-t}$.  Fixed $t$ gives
\eqref{eq:multiplier-covariance-rate}; the sample-centering term
$\bar U\bar U^\top$ is smaller by the score-size bound.
\end{proof}

\begin{corollary}[Gaussian multiplier]
\label{cor:multiplier}
Under the assumptions of Theorem~\ref{thm:matched-chaos}, including
$(sL)^4=o(n)$, the conditional law of
\begin{equation}\label{eq:multiplier-statistic}
        \frac{T_C(Z_1^\#,Z_2^\#)}{\sigma_0\sigma_rsL}
\end{equation}
has $d_3$ distance $o_{\Prob}(1)$ from the law of the normalized searched
displacement in \eqref{eq:smooth-approximation}.
\end{corollary}

\begin{proof}
Lemma~\ref{lem:multiplier-covariance} gives $\eta_n=o_{\Prob}(1)$.
Gaussian interpolation between $\Sigma_U$ and $\widehat\Sigma_U$ uses the
smooth supremum from the proof of Theorem~\ref{thm:matched-chaos}.  With
$\pi_S$ denoting the softmax weights, its Hessian is
\begin{equation*}
        \nabla^2 f_\lambda
        =\sum_S\pi_S\nabla^2\widetilde Q_S
        +\lambda\operatorname{Cov}_\pi(\nabla\widetilde Q_S).
\end{equation*}
Differentiating the interpolated expectation of
$\varphi\{f_\lambda/(sL)\}$ gives
\begin{equation}\label{eq:multiplier-interpolation}
\begin{split}
        \frac1{2sL}\E\left[
        \varphi'\operatorname{tr}\{(\widehat\Sigma_U-\Sigma_U)
        \nabla^2f_\lambda\}\right]
        +\frac1{2(sL)^2}\E\left[
        \varphi''(\nabla f_\lambda)^\top
        (\widehat\Sigma_U-\Sigma_U)\nabla f_\lambda\right].
\end{split}
\end{equation}
The supportwise Hessian is confined to one $s$-set, and every difference
of two softmax gradients is confined to the two score channels over the
union of two such sets.  The definition of $\eta_n$, the score-size bound,
and \eqref{eq:multiplier-interpolation} therefore bound the normalized
derivative by $O_{\Prob}(\eta_n+\eta_n\lambda)$.  The smoothing error is
$1/\lambda$, so $\lambda=\eta_n^{-1/2}$ gives a normalized $d_3$ error
$O_{\Prob}(\sqrt{\eta_n})$.  Combining this interpolation bound with
Theorem~\ref{thm:matched-chaos} proves the corollary.
\end{proof}

For jointly Gaussian rows, the score covariance is
\begin{equation}\label{eq:gaussian-score-covariance}
        \Sigma_U=
        \begin{pmatrix}
        \sigma_0^2C+dd^\top&hd^\top\\
        dh^\top&\sigma_r^2C+hh^\top
        \end{pmatrix}.
\end{equation}
Gaussian regression then couples the empirical and matched scores directly
under $sL=o(n)$.

\begin{corollary}[Gaussian rows]
\label{cor:gaussian}
If $(x_0^{(i)},r^{(i)},X_{-0}^{(i)})$ is jointly Gaussian, then under
\eqref{eq:proxy-energy} and $sL=o(n)$,
\begin{equation*}
        \sup_{S\subseteq[p],\,|S|=s}\widehat\beta_{0,S}
        =\widehat\beta_{0,\emptyset}
        +\frac{T_C(Z_1,Z_2)}{n\sigma_0^2}
        +O_{\Prob}\!\left(
        \frac{\sigma_r}{\sigma_0}\delta_{s,n,p}^3
        \right),
\end{equation*}
on an extension of the data probability space, with $\Sigma_U$ given by
\eqref{eq:gaussian-score-covariance}.
\end{corollary}

\begin{proof}
Gaussian regression writes the control row as a sum of its projections
onto $x_0^{(i)}$ and $r^{(i)}$ and an independent Gaussian remainder.
Conditional on the two scalar sample vectors, the remainder contributes a
matrix Gaussian whose $2\times2$ column covariance is their sample Gram
matrix.  Replacing that Gram matrix by the identity and coupling the three
scalar quadratic sums to Gaussian variables costs
$O_{\Prob}(\delta_{s,n,p})$ in the standardized sparse score norm.
The Lipschitz bound for \eqref{eq:search-functional}, together with
\eqref{eq:score-size}, makes the coupling error smaller than the block-Gram
error already present in \eqref{eq:empirical-chaos}.
\end{proof}

\section{Marginal increments and greedy search}

Evaluating $T_C(a,b)$ by enumeration is combinatorial for a general
correlation matrix.  Its exact one-coordinate increment gives a greedy
search that reduces to top-$s$ sorting when $C=I$.  For $j\notin S$,
define the conditional variance and scores
\begin{equation}\label{eq:conditional-scores}
\begin{aligned}
        c_{j\mid S}
        &=1-C_{jS}C_{SS}^{-1}C_{Sj},\\
        a_{j\mid S}
        &=a_j-C_{jS}C_{SS}^{-1}a_S,\\
        b_{j\mid S}
        &=b_j-C_{jS}C_{SS}^{-1}b_S.
\end{aligned}
\end{equation}
Block inversion gives the marginal increment
\begin{equation}
        \Delta_j(S;a,b)
        :=Q_{S\cup\{j\}}(a,b)-Q_S(a,b)
        =-\frac{a_{j\mid S}b_{j\mid S}}{c_{j\mid S}}.
\end{equation}
Conditional-product greedy starts with $S_0=\emptyset$ and, for
$t=1,\ldots,s$, sets
\begin{equation}
        j_t\in\mathop{\arg\max}_{j\notin S_{t-1}}
        \Delta_j(S_{t-1};a,b),
        \qquad
        S_t=S_{t-1}\cup\{j_t\}.
\end{equation}
With $C$ fixed, write $S_s(a,b)$ for the support returned by the rule.
For known $C$, rank-one updates of \eqref{eq:conditional-scores} avoid
enumerating the $\binom ps$ supports.  When $C=I$, the rule selects the
largest $s$ products $-a_jb_j$.

The relevant departure from separability is
\begin{equation}\label{eq:marginal-oscillation}
\begin{split}
        \operatorname{Osc}_s(a,b;C)
        :=\max_{\substack{j_1,\ldots,j_s\in[p]\\
        \text{distinct}}}
        \sum_{t=1}^s
        \max_{\substack{R,R'\subseteq[p]\setminus\{j_t\}\\
        |R|=|R'|=t-1}}
        \left|\Delta_{j_t}(R;a,b)-\Delta_{j_t}(R';a,b)\right|.
\end{split}
\end{equation}
It compares the same coordinate across two histories of equal size.  The
common level of the marginals may therefore change with the search depth.

\begin{theorem}[Equal-depth marginal stability]
\label{thm:marginal-stability}
For every $a,b\in\R^p$, conditional-product greedy satisfies
\begin{equation}
        0\le T_C(a,b)-Q_{S_s(a,b)}(a,b)
        \le\operatorname{Osc}_s(a,b;C).
\end{equation}
\end{theorem}

\begin{proof}
Fix an optimal support $S^\star$.  Order its elements
$j_1^\star,\ldots,j_s^\star$ backward so that
$j_t^\star\notin S_{t-1}$.  To see that this is possible, suppose the
positions after $t$ have been filled.  At least $s-t+1$ elements of
$S^\star$ lie outside the $(t-1)$-element set $S_{t-1}$, while only $s-t$
elements have already been used.

Put $S_t^\star=\{j_1^\star,\ldots,j_t^\star\}$.  The greedy rule and the
availability of $j_t^\star$ give
\begin{equation*}
        \Delta_{j_t}(S_{t-1};a,b)
        \ge\Delta_{j_t^\star}(S_{t-1};a,b).
\end{equation*}
Telescoping both support values now yields
\begin{equation*}
\begin{split}
        T_C(a,b)-Q_{S_s(a,b)}(a,b)
        &\le\sum_{t=1}^s\left\{
        \Delta_{j_t^\star}(S_{t-1}^\star;a,b)
        -\Delta_{j_t^\star}(S_{t-1};a,b)
        \right\}\\
        &\le\operatorname{Osc}_s(a,b;C).
\end{split}
\end{equation*}
The lower bound follows because $S_s(a,b)$ is one of the supports in the
definition of $T_C(a,b)$.
\end{proof}

\begin{corollary}[Stochastic greedy approximation]
\label{cor:stochastic-search}
Suppose the assumptions of Theorem~\ref{thm:matched-chaos} hold.  If
\begin{equation}
        \operatorname{Osc}_s(Z_{1n},Z_{2n};C)
        =o_{\Prob}(\sigma_0\sigma_rsL),
\end{equation}
then
\begin{equation}\label{eq:empirical-greedy}
        \sup_{|S|=s}\widehat\beta_{0,S}
        =\widehat\beta_{0,\emptyset}
        +\frac{Q_{S_s(Z_{1n},Z_{2n})}(Z_{1n},Z_{2n})}
        {n\sigma_0^2}
        +o_{\Prob}\!\left(
        \frac{\sigma_r}{\sigma_0}\frac{sL}{n}
        \right).
\end{equation}

If $(sL)^4=o(n)$ and
\begin{equation}\label{eq:matched-marginal-stability}
        \operatorname{Osc}_s(Z_1,Z_2;C)
        =o_{\Prob}(\sigma_0\sigma_rsL),
\end{equation}
then the coupling in Theorem~\ref{thm:matched-chaos} satisfies
\begin{equation}\label{eq:algorithmic-approximator}
        \sup_{|S|=s}\widehat\beta_{0,S}
        =\widehat\beta_{0,\emptyset}
        +\frac{Q_{S_s(Z_1,Z_2)}(Z_1,Z_2)}{n\sigma_0^2}
        +o_{\Prob}\!\left(
        \frac{\sigma_r}{\sigma_0}\frac{sL}{n}
        \right).
\end{equation}
An independent matched Gaussian draw also gives
\begin{equation}\label{eq:algorithmic-distribution}
\begin{split}
        d_3\!\left(
        \frac{n\sigma_0}{\sigma_rsL}
        \left\{
        \sup_{|S|=s}\widehat\beta_{0,S}
        -\widehat\beta_{0,\emptyset}
        \right\},
        \frac{Q_{S_s(Z_1,Z_2)}(Z_1,Z_2)}
        {\sigma_0\sigma_rsL}
        \right)=o(1).
\end{split}
\end{equation}
The matched-score conclusions \eqref{eq:algorithmic-approximator} and
\eqref{eq:algorithmic-distribution} also hold under
Corollary~\ref{cor:gaussian}, provided
\eqref{eq:matched-marginal-stability} holds, with $sL=o(n)$.

If Corollary~\ref{cor:multiplier} applies, write $\Prob^\#$ for
conditional probability given the data.  If
\begin{equation}\label{eq:multiplier-marginal-stability}
        \Prob^\#\!\left\{
        \operatorname{Osc}_s(Z_1^\#,Z_2^\#;C)
        >\epsilon\sigma_0\sigma_rsL\right\}
        =o_{\Prob}(1)
        \qquad(\epsilon>0),
\end{equation}
then the conditional version of \eqref{eq:algorithmic-distribution}
holds with $Q_{S_s(Z_1^\#,Z_2^\#)}(Z_1^\#,Z_2^\#)$.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:marginal-stability} gives the required
$o_{\Prob}(\sigma_0\sigma_rsL)$ regret for each score pair.  Combining it
with
\eqref{eq:empirical-chaos} proves \eqref{eq:empirical-greedy}, since
$\delta_{s,n,p}^3=o(sL/n)$.  Combining it instead with
\eqref{eq:coupled-approximation} proves
\eqref{eq:algorithmic-approximator}.  For the distributional statement,
every test function defining $d_3$ is bounded and Lipschitz, so replacing
$T_C$ by the greedy value costs $o(1)$.  The same pointwise argument under
conditional probability proves the multiplier statement.
\end{proof}

Local isotropy is one way to verify marginal stability.  Define
\begin{equation}
        \rho_s(C)
        =\sup_{|J|\le2s}\norm{C_{JJ}-I}_{\mathrm{op}}.
\end{equation}

\begin{corollary}[Local isotropy]
If $\rho=\rho_s(C)\le1/2$, then
\begin{equation}\label{eq:isotropy-oscillation}
        \operatorname{Osc}_s(a,b;C)
        \lesssim
        (\rho\sqrt{s}+s\rho^2)
        \norm{a}_{C,s}\norm{b}_{C,s}.
\end{equation}
Consequently, $\rho_s(C)\sqrt{s}\to0$ verifies the marginal-stability
conditions in Corollary~\ref{cor:stochastic-search} for the empirical,
matched, and conditional multiplier scores.
\end{corollary}

\begin{proof}
For $j\notin S$, put
\begin{equation*}
        u=C_{jS}C_{SS}^{-1}a_S,
        \qquad
        v=C_{jS}C_{SS}^{-1}b_S,
        \qquad
        \ell=C_{jS}C_{SS}^{-1}C_{Sj}.
\end{equation*}
Writing $A=\norm{a}_{C,s}$ and $B=\norm{b}_{C,s}$, Cauchy--Schwarz and
local isotropy give
\begin{equation*}
        \ell\le\frac{\rho^2}{1-\rho},
        \qquad
        |u|\le\sqrt\ell\,A,
        \qquad
        |v|\le\sqrt\ell\,B.
\end{equation*}
The exact identity
\begin{equation}
        \Delta_j(S;a,b)+a_jb_j
        =\frac{a_jv+b_ju-uv-\ell a_jb_j}{1-\ell}
\end{equation}
therefore implies, uniformly over $|S|<s$,
\begin{equation*}
        |\Delta_j(S;a,b)+a_jb_j|
        \lesssim
        \rho(|a_j|B+|b_j|A)
        +\rho^2(AB+|a_jb_j|).
\end{equation*}

For distinct $j_1,\ldots,j_s$ and $J=\{j_1,\ldots,j_s\}$,
\begin{equation*}
\begin{aligned}
        \sum_{t=1}^s|a_{j_t}|
        &\le\sqrt{s(1+\rho)}A,
        &\sum_{t=1}^s|b_{j_t}|
        &\le\sqrt{s(1+\rho)}B,\\
        \sum_{t=1}^s|a_{j_t}b_{j_t}|
        &\le(1+\rho)AB.
\end{aligned}
\end{equation*}
Summing the two leakage bounds in each equal-depth difference proves
\eqref{eq:isotropy-oscillation}.

Lemma~\ref{lem:sparse-score-moments} gives
\begin{equation*}
\begin{aligned}
        \norm{Z_{1n}}_{C,s},\ \norm{Z_1}_{C,s}
        &=O_{\Prob}(\sigma_0\sqrt{sL}),\\
        \norm{Z_{2n}}_{C,s},\ \norm{Z_2}_{C,s}
        &=O_{\Prob}(\sigma_r\sqrt{sL}).
\end{aligned}
\end{equation*}
Lemma~\ref{lem:multiplier-covariance} and the same conditional Gaussian
net argument give the corresponding conditional bounds for $Z^\#$ in
outer probability.  Since $s\rho^2=(\rho\sqrt{s})^2$, substitution in
\eqref{eq:isotropy-oscillation} proves the final claim.
\end{proof}

A complementary stochastic argument uses a uniform supply of fresh
coordinates at every search state.

\begin{theorem}[Fresh marginals on a sparse covariance graph]
\label{thm:fresh-marginals}
Suppose the matched score is centered, so $d=h=0$, and
\begin{equation}\label{eq:sparse-score-covariance}
        Z\sim N\!\left(0,
        \begin{pmatrix}
        \sigma_0^2C&\gamma\sigma_0\sigma_rC\\
        \gamma\sigma_0\sigma_rC&\sigma_r^2C
        \end{pmatrix}\right),
        \qquad -1<\gamma\le1-c_\gamma
\end{equation}
for a constant $c_\gamma>0$.  Join $j$ and $k$ when $C_{jk}\ne0$, and let
$D_C$ be the maximum degree of this graph.  If
\begin{equation*}
        L\longrightarrow\infty,
        \qquad
        \log(D_C+1)=o(L),
\end{equation*}
then
\begin{equation}\label{eq:fresh-greedy}
        T_C(Z_1,Z_2)
        -Q_{S_s(Z_1,Z_2)}(Z_1,Z_2)
        =o_{\Prob}(\sigma_0\sigma_rsL).
\end{equation}
\end{theorem}

\begin{proof}
Put $\tau=(1-\gamma)\sigma_0\sigma_r$.  For independent standard
Gaussians $\xi$ and $\eta$, a singleton marginal has the representation
\begin{equation}\label{eq:product-tail}
        -Z_{1j}Z_{2j}
        \overset{d}=\frac{\sigma_0\sigma_r}{2}
        \left\{(1-\gamma)\xi^2-(1+\gamma)\eta^2\right\}.
\end{equation}
It follows that, uniformly for large $x/\tau$ and with constants depending
only on $c_\gamma$,
\begin{equation*}
        \Prob(-Z_{1j}Z_{2j}>x)
        \gtrsim (x/\tau)^{-1/2}\exp(-x/\tau).
\end{equation*}
Indeed, on $|\eta|\le1$ it is enough that
$\xi^2>2x/\tau+(1+\gamma)/(1-\gamma)$, and the second term is bounded by
$(2-c_\gamma)/c_\gamma$.

Fix $S$ with $|S|<s$.  Deleting $S$ and its neighbors removes at most
$(D_C+1)s$ vertices.  The assumptions imply $(D_C+1)s/p\to0$, so the
remaining graph has an independent set $I_S$ of size at least
$p/\{2(D_C+1)\}$ for all sufficiently large $p$.  For every
$j\in I_S$, $C_{jS}=0$, so
\begin{equation*}
        \Delta_j(S;Z_1,Z_2)=-Z_{1j}Z_{2j}.
\end{equation*}
The coordinate pairs indexed by $I_S$ are independent by
\eqref{eq:sparse-score-covariance}.

Set
\begin{equation*}
        \epsilon
        =\left\{
        \frac{1+\log(D_C+1)+\log L}{L}
        \right\}^{1/2}.
\end{equation*}
Then $\epsilon\to0$,
$e^{\epsilon L}/\{(D_C+1)L^{3/2}\}\to\infty$, and
$\log(1/\epsilon)=o(\epsilon L)$.
For $x=(1-\epsilon)\tau L$, the probability that every fresh marginal at
a fixed state is below $x$ is at most
\begin{equation*}
        \exp\left\{-\frac{c_1s e^{\epsilon L}}
        {(D_C+1)\sqrt L}\right\}.
\end{equation*}
There are at most $e^{2sL}$ sets of size less than $s$.  A union bound
therefore shows that, with probability tending to one, every search state
has a marginal at least $(1-\epsilon)\tau L$.  On this event,
\begin{equation}\label{eq:fresh-lower}
        Q_{S_s(Z_1,Z_2)}(Z_1,Z_2)
        \ge(1-\epsilon)\tau sL.
\end{equation}

For a fixed support, whitening by $C_{SS}^{-1/2}$ writes $Q_S(Z_1,Z_2)$
as a sum of $s$ independent copies of the variable in
\eqref{eq:product-tail}.  Its moment-generating function satisfies
\begin{equation*}
        \E e^{t(-Z_{1j}Z_{2j})}
        =\{(1-\tau t)
        [1+(1+\gamma)\sigma_0\sigma_rt]\}^{-1/2}
        \le(1-\tau t)^{-1/2}.
\end{equation*}
Taking $t=(1-\epsilon/4)/\tau$, applying Chernoff's inequality, and
using $\binom ps\le e^{sL}$ give
\begin{equation}\label{eq:fresh-upper}
        T_C(Z_1,Z_2)\le(1+\epsilon)\tau sL
\end{equation}
with probability tending to one.  The term
$\tfrac{s}{2}\log(4/\epsilon)$ from the moment-generating function is
$o(\epsilon sL)$ by construction.  Combining
\eqref{eq:fresh-lower} and \eqref{eq:fresh-upper}, and using
$\tau\asymp\sigma_0\sigma_r$, proves \eqref{eq:fresh-greedy}.
\end{proof}

Under the assumptions of Theorems~\ref{thm:matched-chaos} and
\ref{thm:fresh-marginals}, including $(sL)^4=o(n)$, the greedy value may
therefore replace $T_C(Z_1,Z_2)$ in
\eqref{eq:algorithmic-approximator} and
\eqref{eq:algorithmic-distribution}.  The covariance condition is imposed
on $\Sigma_U$ and hence on fourth moments of the original rows.

\section{Manuscript assumptions}

The whitened block-Gram event \eqref{eq:block-gram} is a relative RIP over
$2s$-coordinate blocks.  Its scale $\delta_{s,n,p}$ controls the empirical
reduction \eqref{eq:empirical-chaos}.  Replacing $Z_n$ by $Z$ requires
$(sL)^4=o(n)$, while the direct coupling for Gaussian rows requires only
$sL=o(n)$.  Greedy search needs a separate property of the score
marginals.  Equal-depth marginal stability is a score-adaptive sufficient
condition.  The local-isotropy bound verifies it for raw greedy under
$\rho_s(C)\sqrt{s}\to0$, a stronger rate than the manuscript's
$\rho_s(C)=o(1)$.  The condition may instead be checked directly, while
Theorem~\ref{thm:fresh-marginals} uses a uniform supply of fresh marginals.
The relative RIP and these computational conditions do not imply one
another.

Under $\rho_s(C)=o(1)$, coordinatewise and invariant proxy bounds are
related by
\begin{equation}\label{eq:coordinate-to-invariant}
        \norm{v}_{C,2s}
        \le\frac{\sqrt{2s}\norm{v}_\infty}
        {\sqrt{1-\rho_s(C)}}
        \qquad(v\in\R^p).
\end{equation}
Thus the manuscript's caps on $d/\sigma_0$ and
$C\beta_{-0}/\varsigma$ at order $\sqrt{L/n}$ imply the corresponding
covariance-normalized bounds at order $\delta_{s,n,p}$.  The invariant
bounds permit individual coordinates to be larger.

The manuscript's outcome-energy quantities are
\begin{equation*}
        \varsigma^2
        =\Var(y^{(i)}-\beta_0x_0^{(i)})
        =\beta_{-0}^\top C\beta_{-0}+\Var(\varepsilon^{(i)}),
        \qquad
        \Psi_s^2
        =\max_{|J|\le s}
        (C\beta_{-0})_J^\top C_{JJ}^{-1}
        (C\beta_{-0})_J.
\end{equation*}
A partition of each $2s$-set into two $s$-sets gives
\begin{equation*}
        \norm{C\beta_{-0}}_{C,2s}^2
        \le\frac{2\{1+\rho_s(C)\}}{1-\rho_s(C)}\Psi_s^2.
\end{equation*}
Thus $\Psi_s\lesssim\varsigma\delta_{s,n,p}$ yields the required bound
for $C\beta_{-0}$.  If, in addition,
$|d^\top\beta_{-0}|=o(\sigma_0\varsigma\delta_{s,n,p})$, then
\eqref{eq:residual-outcome}, $sL=o(n)$, and the first bound in
\eqref{eq:proxy-energy} give
\begin{equation*}
        \frac{\sigma_r}{\varsigma}\longrightarrow1,
        \qquad
        \norm{h-C\beta_{-0}}_{C,2s}
        =o(\varsigma\delta_{s,n,p}^2).
\end{equation*}
Thus either the coordinatewise cap on $C\beta_{-0}/\varsigma$ or the
$\Psi_s$ condition, together with small alignment and the first bound in
\eqref{eq:proxy-energy}, implies the second bound.

The covariance $\Sigma_U$ retains fourth-moment dependence, so sparsity of
$C$ alone does not give the separable score covariance in
\eqref{eq:sparse-score-covariance}.  For jointly Gaussian rows with
$d=h=0$, \eqref{eq:gaussian-score-covariance} has that form with
$\gamma=0$.  For general sub-Gaussian rows it is an additional structural
condition on fourth moments.  The sample multiplier covariance is
generally dense, so its greedy approximation is governed separately by
\eqref{eq:multiplier-marginal-stability}.
