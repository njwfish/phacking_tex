\author{Th\'eo Voldoire and Nic Fishman}

\begin{abstract}
A covariance-normalized bilinear search has an exact product form for
every positive semidefinite, possibly singular covariance matrix: each
coordinate contributes the product of the two scores after residualizing
on the coordinates selected before it.  A fixed polarization separates
these products into an active and a passive projection energy.  Forward
selection of the active energy then gives a polynomial-time product path with
an exact realized error certificate controlled by a scale-invariant
projection ratio.  Applied to the empirical Gram matrix, the same
construction gives a deterministic finite-sample regression certificate
from the data and a supplied design bound.  Relative restricted isometry
transfers the population projection bound to this empirical certificate.  Under
centered Kronecker Gaussian scores, the passive energy has an exact
chi-square law, and the forward path is first-order accurate for
block-diagonal covariance with many independent blocks even when the
covariance within each block is anisotropic or singular.  Coordinatewise
top-$s$ sorting is the diagonal special case of this sequential product
construction.
\end{abstract}

\section{Conditional products}

Let $C$ be a positive semidefinite matrix.  Discard its zero-variance
coordinates, reindex the remaining coordinates by $[p]=\{1,\ldots,p\}$,
and fix $1\le s\le p$.  Every inverse below acts on the range of the
matrix, or equivalently is its Moore--Penrose inverse in the original
coordinates.  A vector $v\in\R^p$ is compatible with $C$ through order
$2s$ when
\begin{equation}\label{eq:compatibility}
        v_J\in\operatorname{range}(C_{JJ})
        \qquad\text{whenever }|J|\le2s.
\end{equation}
This convention preserves exact linear dependencies among the
coordinates and removes the need for positive definite covariance
blocks.

For compatible $v$ and $|S|\le2s$, define its projection energy by
\begin{equation}\label{eq:projection-energy}
        E_S(v)=v_S^\top C_{SS}^{-1}v_S,
        \qquad
        \norm{v}_{C,s}^2=\max_{|S|=s}E_S(v).
\end{equation}
For compatible $a,b\in\R^p$, put
\begin{equation}\label{eq:bilinear-search}
        Q_S(a,b)=-a_S^\top C_{SS}^{-1}b_S,
        \qquad
        T_C(a,b)=\max_{|S|=s}Q_S(a,b).
\end{equation}

If $j\notin S$, let
\begin{equation*}
        c_{j\mid S}=C_{jj}-C_{jS}C_{SS}^{-1}C_{Sj}.
\end{equation*}
When $c_{j\mid S}>0$, define the standardized residual score
\begin{equation}\label{eq:residual-score}
        v_{j\mid S}
        =\frac{v_j-C_{jS}C_{SS}^{-1}v_S}{c_{j\mid S}^{1/2}},
\end{equation}
and set $v_{j\mid S}=0$ when $c_{j\mid S}=0$.  Compatibility makes the
residual numerator zero in the latter case.

\begin{theorem}[Conditional product identity]
\label{thm:conditional-product}
For every compatible $a,b$, every $|S|<2s$, and every $j\notin S$,
\begin{equation}\label{eq:product-increment}
        Q_{S\cup\{j\}}(a,b)-Q_S(a,b)
        =-a_{j\mid S}b_{j\mid S}.
\end{equation}
Consequently, if $j_1,\ldots,j_s$ are distinct and
$J_t=\{j_1,\ldots,j_t\}$, then
\begin{equation}\label{eq:product-path}
        Q_{J_s}(a,b)
        =-\sum_{t=1}^s
        a_{j_t\mid J_{t-1}}b_{j_t\mid J_{t-1}},
        \qquad J_0=\emptyset.
\end{equation}
The sum depends only on $J_s$, not on the order used to reach it.
\end{theorem}

\begin{proof}
When $c_{j\mid S}>0$, support compression and block inversion give
\begin{equation*}
\begin{split}
        &a_{S\cup\{j\}}^\top
        C_{S\cup\{j\},S\cup\{j\}}^{-1}b_{S\cup\{j\}}
        -a_S^\top C_{SS}^{-1}b_S\\
        &\qquad=
        \frac{(a_j-C_{jS}C_{SS}^{-1}a_S)
        (b_j-C_{jS}C_{SS}^{-1}b_S)}{c_{j\mid S}}
        =a_{j\mid S}b_{j\mid S}.
\end{split}
\end{equation*}
If $c_{j\mid S}=0$, the coordinate adds no direction to the selected
span and both sides of \eqref{eq:product-increment} vanish.  Telescoping
proves \eqref{eq:product-path}, whose terminal value depends only on the
selected support.
\end{proof}

When $C$ is diagonal, the residual products do not depend on the path,
and $T_C(a,b)$ is the sum of the $s$ largest values among
$-a_jb_j/C_{jj}$.  Correlation makes the factors history-dependent but
does not destroy their exact product form.  Selecting the largest current
residual product is therefore the literal extension of top-$s$ sorting.
The path below uses a polarized projection gain instead, which permits a
computable regret certificate.

\section{Forward product search}

The product path becomes computationally useful after one fixed
polarization.  For positive scales $\sigma_a,\sigma_b$, define the active
and passive contrasts
\begin{equation}\label{eq:polarization}
        z_{\mathrm a}=\frac1{\sqrt2}
        \left(\frac a{\sigma_a}-\frac b{\sigma_b}\right),
        \qquad
        z_{\mathrm p}=\frac1{\sqrt2}
        \left(\frac a{\sigma_a}+\frac b{\sigma_b}\right).
\end{equation}
Residualization commutes with this rotation, so
\begin{equation}\label{eq:polarized-product}
\begin{split}
        -a_{j\mid S}b_{j\mid S}
        &=\frac{\sigma_a\sigma_b}{2}
        \{(z_{\mathrm a})_{j\mid S}^2-(z_{\mathrm p})_{j\mid S}^2\},\\
        Q_S(a,b)
        &=\frac{\sigma_a\sigma_b}{2}
        \{E_S(z_{\mathrm a})-E_S(z_{\mathrm p})\}.
\end{split}
\end{equation}

The forward guarantee is governed by conditional rather than marginal
conditioning.  For disjoint $S,R\subseteq[p]$, define
\begin{equation}\label{eq:conditional-covariance}
        C_{R\mid S}
        =C_{RR}-C_{RS}C_{SS}^{-1}C_{SR}.
\end{equation}
The projection ratio through order $s$ is
\begin{equation}\label{eq:residual-ratio}
\begin{split}
        \mu_s(C)=
        \inf_{S,R,u}
        \frac{u^\top\operatorname{diag}(C_{R\mid S})^{-1}u}
        {u^\top C_{R\mid S}^{-1}u},
\end{split}
\end{equation}
where the infimum is over disjoint sets with $|S|\le s$,
$1\le|R|\le s$, and nonzero
$u\in\operatorname{range}(C_{R\mid S})$; pairs with
$C_{R\mid S}=0$ are omitted, and an empty infimum is one.  Equivalently,
\eqref{eq:residual-ratio} is the infimum of the least positive
eigenvalues of the conditional correlation matrices obtained from
\eqref{eq:conditional-covariance} after zero-variance coordinates are
removed.  It is the largest covariance-only constant in the
joint-to-coordinate projection inequality used below: for every admissible
$u$, the compatible score
$v=C_{\cdot R}C_{R\mid S}^{-1}u$ has conditional residual $u$.  Thus every
direction in the infimum is attained.  The ratio is
invariant under coordinatewise rescaling and remains meaningful when
$C$ is singular.  If every correlation-normalized block of order at most
$2s$ is nonsingular and has ordinary condition number at most $K$, then
\begin{equation}\label{eq:conditioning-implies-ratio}
        \mu_s(C)\ge K^{-1}.
\end{equation}
Indeed, after rescaling the coordinates, $C_{R\mid S}^{-1}$ is a
principal block of $C_{S\cup R,S\cup R}^{-1}$.  Thus the eigenvalues of
$C_{R\mid S}$ lie between those of its parent block.  Its diagonal
entries are bounded by the same upper eigenvalue, so its
correlation-normalized least eigenvalue is at least $K^{-1}$.

Fix a supplied lower bound $0<\mu\le\mu_s(C)$.  Starting from
$J_0=\emptyset$, the forward active path chooses
\begin{equation}\label{eq:forward-path}
        j_t\in\mathop{\arg\max}_{j\notin J_{t-1}}
        (z_{\mathrm a})_{j\mid J_{t-1}}^2,
        \qquad
        J_t=J_{t-1}\cup\{j_t\},
        \qquad t=1,\ldots,s,
\end{equation}
with ties resolved by a fixed order.  For a finite family, braces with
subscript $(\ell)$ denote its $\ell$th largest member.  Along this path,
set
\begin{equation}\label{eq:projection-upper-certificate}
\begin{split}
        \mathcal U_{C,\mu}(v;J_\bullet)
        =\min_{0\le t\le s}\bigg\{
        E_{J_t}(v)
        +\frac1\mu
        \sum_{\ell=1}^{s\wedge(p-t)}
        \{v_{j\mid J_t}^2:j\notin J_t\}_{(\ell)}
        \bigg\}.
\end{split}
\end{equation}
Write the product-gap certificate as
\begin{equation}\label{eq:product-gap-certificate}
        \mathcal D_{C,\mu}(z_{\mathrm a},z_{\mathrm p};J_\bullet)
        =\mathcal U_{C,\mu}(z_{\mathrm a};J_\bullet)-E_{J_s}(z_{\mathrm a})+E_{J_s}(z_{\mathrm p}).
\end{equation}

\begin{theorem}[Forward product-search guarantee]
\label{thm:projection-certified-products}
Let $a,b$ be compatible, form $z_{\mathrm a},z_{\mathrm p}$ by
\eqref{eq:polarization}, and let $J_\bullet$ follow
\eqref{eq:forward-path}.  For every compatible $v$,
\begin{equation}\label{eq:projection-upper-bound}
        \mathcal U_{C,\mu}(v;J_\bullet)\ge\norm{v}_{C,s}^2.
\end{equation}
The active gap satisfies, for $0\le t\le s$,
\begin{equation}\label{eq:forward-active-gap}
        0\le\norm{z_{\mathrm a}}_{C,s}^2-E_{J_t}(z_{\mathrm a})
        \le\left(1-\frac\mu s\right)^t
        \norm{z_{\mathrm a}}_{C,s}^2.
\end{equation}
The returned conditional-product sum has the realized certificate
\begin{equation}\label{eq:product-certificate}
\begin{split}
        0&\le T_C(a,b)-Q_{J_s}(a,b)\\
        &\le\frac{\sigma_a\sigma_b}{2}
        \mathcal D_{C,\mu}(z_{\mathrm a},z_{\mathrm p};J_\bullet).
\end{split}
\end{equation}
In particular, the right side bounds the regret of the sequential product
in \eqref{eq:product-path}.
\end{theorem}

\begin{proof}
For disjoint $S,R$, write
\begin{equation*}
        u=v_R-C_{RS}C_{SS}^{-1}v_S.
\end{equation*}
Block inversion on the compressed range gives
\begin{equation}\label{eq:joint-projection-gain}
\begin{split}
        E_{S\cup R}(v)-E_S(v)
        &=u^\top C_{R\mid S}^{-1}u,\\
        \sum_{j\in R}v_{j\mid S}^2
        &=u^\top\operatorname{diag}(C_{R\mid S})^{-1}u\\
        \ge\mu\{E_{S\cup R}(v)-E_S(v)\}.
\end{split}
\end{equation}
The final inequality is the definition of $\mu_s(C)$.

Let $A$ maximize $E_A(v)$ over $|A|=s$, and take $R=A\setminus J_t$ in
\eqref{eq:joint-projection-gain}.  Projection energy is monotone, so
\begin{equation*}
        \norm{v}_{C,s}^2-E_{J_t}(v)
        \le\frac1\mu
        \sum_{\ell=1}^{s\wedge(p-t)}
        \{v_{j\mid J_t}^2:j\notin J_t\}_{(\ell)}.
\end{equation*}
This proves \eqref{eq:projection-upper-bound}.  With $v=z_{\mathrm a}$, the largest
residual gain is at least $\mu/s$ times the current active gap.  Iteration
proves \eqref{eq:forward-active-gap}.

Polarization also gives the exact error budget
\begin{equation*}
\begin{split}
        &\{T_C(a,b)-Q_{J_s}(a,b)\}
        +\left\{\frac{\sigma_a\sigma_b}{2}
        \norm{z_{\mathrm a}}_{C,s}^2-T_C(a,b)\right\}\\
        &\qquad=\frac{\sigma_a\sigma_b}{2}
        \left\{\norm{z_{\mathrm a}}_{C,s}^2-E_{J_s}(z_{\mathrm a})+E_{J_s}(z_{\mathrm p})\right\}.
\end{split}
\end{equation*}
Both terms on the left are nonnegative.  Dropping the second and applying
\eqref{eq:projection-upper-bound} proves
\eqref{eq:product-certificate}.
\end{proof}

Incremental factorization computes the path and all certificates in
$O(ps^2)$ arithmetic operations and $O(ps)$ working memory beyond access
to $C$.  The path itself requires no value of $\mu$; the supplied lower
bound makes its regret certificate valid.  Obtaining the best possible
sparse lower bound can be combinatorial, so the computational statement
applies when design structure or a conservative spectral bound supplies
$\mu$.

\section{A finite-sample regression certificate}

The projection certificate applies directly to the sample rather than
only to a Gaussian approximation.  Let $x,y\in\R^n$ satisfy
$x^\top x>0$, let $X\in\R^{n\times p}$, and, for each $S$, let
$\hat\beta_{0,S}$ be the coefficient on $x$ in the regression of
$y$ on $(x,X_S)$.  Define
\begin{equation}\label{eq:empirical-objects}
\begin{gathered}
        \hat\beta_{0,\emptyset}=\frac{x^\top y}{x^\top x},
        \qquad
        \hat r=y-\hat\beta_{0,\emptyset}x,
        \qquad
        \hat\sigma_0^2=\frac{x^\top x}{n},\\
        \hat C=\frac{X^\top X}{n},
        \qquad
        a=\frac{X^\top x}{\sqrt n},
        \qquad
        b=\frac{X^\top\hat r}{\sqrt n},
        \qquad
        \hat\sigma_r^2=\frac{\hat r^\top\hat r}{n}.
\end{gathered}
\end{equation}
The two scores are compatible with $\hat C$.  Apply
Theorem~\ref{thm:projection-certified-products} with $C=\hat C$,
any positive scales $\sigma_a,\sigma_b$, and a supplied
$0<\mu\le\mu_s(\hat C)$.  In the rest of this section, $E_S$ is evaluated
at $\hat C$.  Along the resulting path, write
\begin{equation}\label{eq:sample-certificates}
\begin{split}
        \hat{\mathcal D}
        &=\mathcal D_{\hat C,\mu}(z_{\mathrm a},z_{\mathrm p};J_\bullet),\\
        \mathcal U_a&=\mathcal U_{\hat C,\mu}(a;J_\bullet),
        \qquad
        \mathcal U_b=\mathcal U_{\hat C,\mu}(b;J_\bullet).
\end{split}
\end{equation}
When $\hat\sigma_0\hat\sigma_r>0$, the two sample standard
deviations give a canonical choice of the polarization scales.

\begin{theorem}[Finite-sample regression certificate]
\label{thm:observable-regret}
If $\mathcal U_a<n\hat\sigma_0^2$, then every searched denominator is
positive and
\begin{equation}\label{eq:observable-regret}
\begin{split}
        0&\le
        \max_{|S|=s}\hat\beta_{0,S}
        -\hat\beta_{0,J_s}\\
        &\le
        \frac{\sigma_a\sigma_b\hat{\mathcal D}}
        {2n\hat\sigma_0^2}
        +\frac{2\mathcal U_a\sqrt{\mathcal U_a\mathcal U_b}}
        {n\hat\sigma_0^2
        (n\hat\sigma_0^2-\mathcal U_a)}.
\end{split}
\end{equation}
Every quantity on the right is computed from $(x,y,X)$, the chosen
scales, and the supplied design bound $\mu$.
\end{theorem}

\begin{proof}
Let $M_S$ project orthogonally off the columns of $X_S$.  Since
$x^\top\hat r=0$, partitioned regression gives the exact identity
\begin{equation}\label{eq:exact-empirical-ratio}
        \hat\beta_{0,S}-\hat\beta_{0,\emptyset}
        =\frac{Q_S(a,b)}{n\hat\sigma_0^2-E_S(a)}.
\end{equation}
Theorem~\ref{thm:projection-certified-products} gives
$E_S(a)\le\mathcal U_a$ and $E_S(b)\le\mathcal U_b$ for every
size-$s$ support.  Cauchy--Schwarz on the compressed range therefore
gives
\begin{equation*}
        |Q_S(a,b)|\le\sqrt{\mathcal U_a\mathcal U_b}.
\end{equation*}
Uniformly over the searched supports,
\begin{equation}\label{eq:denominator-error}
\begin{split}
        \left|
        \frac{Q_S(a,b)}{n\hat\sigma_0^2-E_S(a)}
        -\frac{Q_S(a,b)}{n\hat\sigma_0^2}
        \right|
        \le
        \frac{\mathcal U_a\sqrt{\mathcal U_a\mathcal U_b}}
        {n\hat\sigma_0^2
        (n\hat\sigma_0^2-\mathcal U_a)}.
\end{split}
\end{equation}
The maximum of $Q_S(a,b)/(n\hat\sigma_0^2)$ is
$T_{\hat C}(a,b)/(n\hat\sigma_0^2)$, and
\eqref{eq:product-certificate} bounds its difference from
$Q_{J_s}(a,b)/(n\hat\sigma_0^2)$ by
$\sigma_a\sigma_b\hat{\mathcal D}/
(2n\hat\sigma_0^2)$.  Applying
\eqref{eq:denominator-error} once to the maximizing support and once to
$J_s$ proves \eqref{eq:observable-regret}.
\end{proof}

The second term in \eqref{eq:observable-regret} is a uniform certified
cost of replacing the support-dependent regression denominators by
$n\hat\sigma_0^2$.  Thus the certificate remains valid when the outcome
signal is nonnegligible and in observational designs; no stochastic
reduction is required to run or verify the search.

The projection ratio is also expressed in the same relative geometry as a
restricted-isometry condition.

\begin{lemma}[Relative RIP transfer]
\label{lem:rip-transfer}
Let $C$ and $\widetilde C$ be positive semidefinite matrices such that,
for every $|J|\le2s$ and some $0\le\epsilon<1$,
\begin{equation}\label{eq:relative-rip}
        (1-\epsilon)C_{JJ}
        \preceq\widetilde C_{JJ}
        \preceq(1+\epsilon)C_{JJ}.
\end{equation}
Then
\begin{equation}\label{eq:rip-ratio-transfer}
        \mu_s(\widetilde C)
        \ge\frac{1-\epsilon}{1+\epsilon}\mu_s(C).
\end{equation}
\end{lemma}

\begin{proof}
For disjoint $S,R$, the Schur complement has the variational
representation
\begin{equation*}
        u^\top C_{R\mid S}u
        =\inf_v
        \begin{pmatrix}v\\u\end{pmatrix}^{\!\top}
        C_{S\cup R,S\cup R}
        \begin{pmatrix}v\\u\end{pmatrix}.
\end{equation*}
Hence \eqref{eq:relative-rip} gives
\begin{equation*}
        (1-\epsilon)C_{R\mid S}
        \preceq\widetilde C_{R\mid S}
        \preceq(1+\epsilon)C_{R\mid S}.
\end{equation*}
The two conditional matrices have the same range and zero diagonal
coordinates.  On their common range, their inverse quadratic forms obey
the reversed inequalities, while their diagonal inverse quadratic forms
obey the corresponding coordinatewise inequalities.  Therefore, for
every admissible nonzero $u$,
\begin{equation*}
        \frac{u^\top
        \operatorname{diag}(\widetilde C_{R\mid S})^{-1}u}
        {u^\top\widetilde C_{R\mid S}^{-1}u}
        \ge\frac{1-\epsilon}{1+\epsilon}
        \frac{u^\top\operatorname{diag}(C_{R\mid S})^{-1}u}
        {u^\top C_{R\mid S}^{-1}u}.
\end{equation*}
Taking the infimum proves \eqref{eq:rip-ratio-transfer}.
\end{proof}

In particular, the relative restricted-isometry event for the empirical
Gram matrix transfers any population lower bound on $\mu_s(C)$ to the
finite-sample certificate.  A bounded sparse condition number is a simple
sufficient condition, but \eqref{eq:residual-ratio} is weaker and allows
exact dependencies.

\section{Stochastic accuracy beyond diagonal covariance}

For random scores and a deterministic scale $\omega_n>0$,
\begin{equation}\label{eq:stochastic-certificate}
        \mathcal D_{C,\mu}(z_{\mathrm a},z_{\mathrm p};J_\bullet)=o_{\Prob}(\omega_n)
\end{equation}
implies
\begin{equation}\label{eq:stochastic-products}
        T_C(a,b)
        =-\sum_{t=1}^s
        a_{j_t\mid J_{t-1}}b_{j_t\mid J_{t-1}}
        +o_{\Prob}(\sigma_a\sigma_b\omega_n).
\end{equation}
Gaussian structure provides useful conditions under which the passive
part of \eqref{eq:stochastic-certificate} is automatically small.

Suppose, for $-1<\gamma<1$, that
\begin{equation}\label{eq:kronecker-scores}
        \begin{pmatrix}a/\sigma_a\\b/\sigma_b\end{pmatrix}
        -\E\begin{pmatrix}a/\sigma_a\\b/\sigma_b\end{pmatrix}
        \sim N\!\left(
        0,
        \begin{pmatrix}1&\gamma\\\gamma&1\end{pmatrix}\otimes C
        \right).
\end{equation}
Then $z_{\mathrm a}$ and $z_{\mathrm p}$ are independent Gaussian vectors with covariances
$(1-\gamma)C$ and $(1+\gamma)C$.

\begin{lemma}[Passive energy]
\label{lem:passive-energy}
Under \eqref{eq:kronecker-scores}, let $J_\bullet$ be any
$z_{\mathrm a}$-measurable ordered path.  Conditionally on $z_{\mathrm a}$,
\begin{equation}\label{eq:passive-chi-square}
        \frac{E_{J_s}(z_{\mathrm p})}{1+\gamma}
        \sim\chi^2_{\operatorname{rank}(C_{J_sJ_s})}
        \left(
        \frac{E_{J_s}(\E z_{\mathrm p})}{1+\gamma}
        \right).
\end{equation}
\end{lemma}

\begin{proof}
Independence makes the conditional law of $z_{\mathrm p}$ equal to its marginal
law.  Along the conditionally fixed path, its nonzero standardized
residuals are independent with variance $1+\gamma$.  Their squared means
sum to $E_{J_s}(\E z_{\mathrm p})$, which proves
\eqref{eq:passive-chi-square}.
\end{proof}

The forward path is first-order accurate for a class that need not be
locally isotropic.  Write $L=\log(ep/s)$.

\begin{theorem}[Forward search under block-diagonal covariance]
\label{thm:block-products}
Suppose \eqref{eq:kronecker-scores} holds with
$\E a=\E b=0$ and, after a permutation, $C$ is block diagonal with $q$
positive-rank blocks.  If
\begin{equation}\label{eq:block-regime}
        s=o(q),
        \qquad
        \log(p/q)=o(L),
\end{equation}
then the forward path in \eqref{eq:forward-path}, which does not require
$\mu$, satisfies
\begin{equation}\label{eq:block-active-gap}
        \norm{z_{\mathrm a}}_{C,s}^2-E_{J_s}(z_{\mathrm a})
        =o_{\Prob}(sL)
\end{equation}
and
\begin{equation}\label{eq:block-product-approximation}
\begin{split}
        T_C(a,b)
        &=-\sum_{t=1}^s
        a_{j_t\mid J_{t-1}}b_{j_t\mid J_{t-1}}
        +o_{\Prob}(\sigma_a\sigma_b sL)\\
        &=\sigma_a\sigma_b(1-\gamma)sL
        +o_{\Prob}(\sigma_a\sigma_b sL).
\end{split}
\end{equation}
No conditioning or full-rank assumption is imposed within the blocks.
\end{theorem}

\begin{proof}
Put $\widetilde z=z_{\mathrm a}/(1-\gamma)^{1/2}$, so
$\widetilde z\sim N(0,C)$.  For every fixed size-$s$ support,
$E_S(\widetilde z)$ is chi-square with at most $s$ degrees of freedom.
A chi-square tail bound and a union over the at most $\binom ps$ supports
give
\begin{equation}\label{eq:block-upper}
        \norm{\widetilde z}_{C,s}^2
        \le\{2+o_{\Prob}(1)\}sL.
\end{equation}

Choose one coordinate from each block.  Their standardized scores are
independent standard Gaussians.  Gaussian tail and binomial concentration
bounds give
\begin{equation}\label{eq:block-lower}
        \sum_{t=1}^s
        \left\{\frac{\widetilde z_j^2}{C_{jj}}:
        j\text{ is a block representative}\right\}_{(t)}
        \ge\{2-o_{\Prob}(1)\}s\log(eq/s)
        =\{2-o_{\Prob}(1)\}sL.
\end{equation}
At the $t$th forward step, at most $t-1$ blocks have been touched.  One
of the blocks carrying the $t$ largest representative squares is still
untouched, and its representative has zero covariance with every
selected coordinate.  Its residual gain is unchanged.  The $t$th
forward gain is therefore at least the $t$th largest representative
square.  Telescoping \eqref{eq:block-lower} and comparing with
\eqref{eq:block-upper} proves \eqref{eq:block-active-gap}.

Lemma~\ref{lem:passive-energy} gives $E_{J_s}(z_{\mathrm p})=O_{\Prob}(s)$, while
\eqref{eq:block-regime} implies $L\to\infty$.  The exact polarized error
budget in the proof of Theorem~\ref{thm:projection-certified-products}
then proves the first equality in
\eqref{eq:block-product-approximation}.  The selected active energy is
$2(1-\gamma)sL+o_{\Prob}(sL)$, so
\eqref{eq:polarized-product} proves the second.
\end{proof}

The diagonal case makes every conditional product static and recovers one
top-$s$ sort.  The block result shows why local isotropy is not the
boundary of product search: general covariance replaces a static list by
a residual product path, and the active projection supplies both the path
