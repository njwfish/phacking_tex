\subsection{Representation Search}

% Note item:
% - move beyond choosing controls
% - allow variable representation / preprocessing choices
% - this changes the available search geometry

\tc{The searches of Sections~\ref{sec:bounds} and~\ref{sec:plugin} select
controls from a pre-determined list, at and below the
isometry scaling $n \asymp s \log(ep/s)$, which leaves aside two
observations. First, a researcher may use pre-processing as a means to
reach statistical significance, combining controls while still reporting
only a few of them. Second, the proportional regime, in which $p/n$ tends
to a constant, leads to qualitatively distinct behaviour in many
settings. Progress on both is made here by enlarging the search space
available to the analyst, who residualizes against any rank-$\le s$
subspace of the control span, the basis search of
\eqref{eq:relaxed-search},
\begin{equation*}
        \hat\beta_{0,s}^\star = \sup_{P \in \mathcal{P}_s} \hat\beta_0^{(P)},
        \qquad
        \hat\beta_0^{(P)}
        = \frac{x_0^\top (I - P)\, y}{x_0^\top (I - P)\, x_0},
\end{equation*}
where the admissible projectors are those the control span can furnish,
\begin{equation*}
        \mathcal{P}_s
        = \big\{ P = P^\top = P^2 :
        \operatorname{Im}(P) \subseteq \operatorname{span}(X_{-0}),\;
        \rank(P) \le s \big\} .
\end{equation*}
This is a strict expansion, and since $P_S \in \mathcal{P}_s$ for every
$S$, coordinate search is dominated pathwise,
$\sup_{|S| = s} \hat\beta_{0,S} \le \hat\beta_{0,s}^\star$. The
representation is basis-independent, and reaches every pre-processing
step that takes linear transforms of the controls and reports $s$
aggregate covariates. However, it does not cover other forms of
pre-processing, non-linear transforms and changes to the treatment
covariate among them; the tools developed here may be adapted to those
cases, which is left to further work and discussed in
Section~\ref{sec:discussion}.}

\tc{In the proportional regime the design of Assumption~\ref{ass:dgp}(i)
is taken in coordinate form, the structure under which the projected
quadratic forms entering the searched ratio self-average to deterministic
limits.}

\begin{assumption}[Continuous-search regime]\label{ass:continuous}
\tc{The following hold.
\begin{enumerate}
\item[(i)] \emph{Design and treatment.} The controls follow the linear
model $X_{-0} = Z\, C^{1/2}$, where $Z \in \R^{n \times p}$ has i.i.d.\
entries that are centered, of unit variance and $K_\psi$-subgaussian, and
the treatment decomposes as $x_0 = X_{-0}\, C^{-1} d + u$, with
$u \in \R^n$ having i.i.d.\ centered $K_\psi$-subgaussian entries of
variance $\sigma_0^2 - d^\top C^{-1} d$, independent of $Z$ and of
$\varepsilon$.
\item[(ii)] \emph{Proportional regime.} The dimensions grow
proportionally, $p / n \to \kappa \in (0,1)$, and the search budget
satisfies $1 \le s \le p$, fixed or growing with $n$ at any rate.
\item[(iii)] \emph{Population regularity.} The covariance $C$ has
eigenvalues in $[\lambda_-, \lambda_+]$ for constants
$0 < \lambda_- \le \lambda_+ < \infty$; the scalars $\sigma_0^2$,
$\sigma_\varepsilon^2$, $\beta_{-0}^\top C \beta_{-0}$,
$d^\top \beta_{-0}$ and $d^\top C^{-1} d$ are bounded and converge as
$n, p \to \infty$; and the treatment keeps variance off the controls,
$d^\top C^{-1} d \le (1 - c_{\mathrm{ov}})\, \sigma_0^2$ for a constant
$c_{\mathrm{ov}} \in (0, 1]$.
\end{enumerate}}
\end{assumption}

\tc{Condition (i) specializes Assumption~\ref{ass:dgp}(i), preserving the
row second moments while adding the independence across coordinates under
which the searched quadratic forms self-average, and it is the only place
that independence is used. Conditions (ii) and (iii) are the usual
hypotheses of proportional asymptotics in random-matrix theory, amounting
to a proportional scaling together with a bounded condition number for
the full covariance of $(x_0, X_{-0})$.}

\subsection{Scalar Limit}

% Note item:
% - scalar limit
% - sharply predictive
% - applies to the relaxed continuous case
% - use as benchmark for both experimental and confounded geometries

\begin{theorem}[Basis search under proportional asymptotics]
\label{thm:continuous}
\tc{Let Assumptions~\ref{ass:dgp}(i) and~\ref{ass:continuous} hold. Then
\begin{equation}\label{eq:continuous-limit}
        \hat\beta_{0,s}^\star
        \;\toP\;
        \beta_0 + q
        + \sqrt{\frac{1}
        {(1-\kappa)\!\left(1 - \frac{d^\top C^{-1} d}{\sigma_0^2}\right)}
        - 1}\;
        \sqrt{\frac{\beta_{-0}^\top C \beta_{-0}
        + \kappa\, \sigma_\varepsilon^2}{4\, \sigma_0^2} - q^2},
\end{equation}}
where $q = \frac{d^\top \beta_{-0}}{2\, \sigma_0^2}$ is the overlap between
the confounding and the outcome signal, half the omitted-variable bias
carried by the no-control estimator, $\hat\beta_{0,\emptyset} \toP
\beta_0 + 2q$.
\end{theorem}

\tc{A first interesting case for this theorem is $\kappa = 0$, in low
dimension, where the limit reads
\begin{equation*}
        \beta_0 + q
        + \frac{R}{\sqrt{1 - R^2}}\;
        \sqrt{\frac{\beta_{-0}^\top C \beta_{-0}}{4\, \sigma_0^2} - q^2},
        \qquad
        R^2 = \frac{d^\top C^{-1} d}{\sigma_0^2},
\end{equation*}
with $R^2$ the share of the treatment variance the controls explain. This
case corresponds to the known extreme bound of
\citet[p.~311]{leamerLeonard1983reporting} and
\citet[eq.~(17)]{breusch1990simplified}, stated as Theorem~1 of
\citet{grangerUhlig1990reasonable}, who maximize the reported coefficient
over all linear combinations of the doubtful controls and obtain a closed
form computable from the short and long regressions alone, already
independent of how many combinations are restricted. The theorem adds
that the same closed form, and the same absence of dependence on the
budget, hold throughout the proportional regime. Finally, under
randomization in low dimension, $d = 0$ and $\kappa = 0$, the whole
expression reduces to $\beta_0$, leaving no potential for hacking.

Turning to the general case $\kappa > 0$, one notices first that $d = 0$
is no longer enough to imply a zero displacement, the supremum being
instead
\begin{equation*}
        \sqrt{\frac{\kappa}{1-\kappa}}\;
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0}
        + \kappa\, \sigma_\varepsilon^2}}{2\, \sigma_0}.
\end{equation*}
First, notice that the factor $1/\sqrt{1-\kappa}$ is the variance
inflation usual in the proportional regime, and that
$\kappa\, \sigma_\varepsilon^2$ is the outcome noise the control span
absorbs, $\sigma_\varepsilon^2$ times the maximum spurious $R^2$ of
\citet{fanShaoZhou2018spurious} taken over the whole span. Both prices
grow with the ratio $p/n$, the analyst accumulating noise with the number
of covariates available, whether or not any of them is reported. Now
comparing against Theorem~\ref{thm:experimental-matching}, which
displaces the coefficient by
$\delta_{s,n,p} \sqrt{\Psi_s^2 + \delta_{s,n,p}^2
(\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2)} / \sigma_0$ with
$\Psi_s^2 \le \beta_{-0}^\top C \beta_{-0}$, the two carry the same
outcome term and differ only in the factor multiplying it,
$\sqrt{s \log(ep/s)/n}$ for coordinate search against $\sqrt\kappa/2$
for basis search. A single manufactured control therefore displaces the
coefficient as much as a coordinate search over a budget of order
$p / \log p$.

Away from these two cases the confounding and the dimension act together,
through the variance inflation and the exploitable outcome energy named
in \eqref{eq:relaxed-rmt}. The expression is heavier than in the special
cases, but may be characterized alternatively as the larger root of a
quadratic equation. Writing $\delta$ for the displacement from $\beta_0$
and $A = d^\top C^{-1} d + \kappa\,(\sigma_0^2 - d^\top C^{-1} d)$ for
the treatment energy lying inside the control span, that equation is
\begin{equation}\label{eq:cont-quadratic}
        \sigma_0^2 (\sigma_0^2 - A)\, \delta^2
        \;-\; (d^\top \beta_{-0}) (\sigma_0^2 - A)\, \delta
        \;+\; \tfrac14 \Big( (d^\top \beta_{-0})^2
        - A \big( \beta_{-0}^\top C \beta_{-0}
        + \kappa\, \sigma_\varepsilon^2 \big) \Big)
        \;=\; 0,
\end{equation}
so that \eqref{eq:continuous-limit} is $\beta_0$ plus its larger root.}

\subsection{The Searched t-Statistic}

% Note item:
% - studentization changes the variational problem itself (rank-0/1/2, not
%   the rank-1 of the coefficient); no one-line transfer as in Section 2.4
% - own proportional-regime limit; natural next benchmark
% AUTHOR NOTE: introduces the continuous t-stat notation
% (\hat t^{(P)}_0, \hat t^\star_{0,s}), the projector analogue of the
% coordinate \hat t_{0,S} of eq:tstat-def. Flag if the intro should own it.

\tc{Searching the studentized statistic over the same projector class,
\begin{equation}\label{eq:cont-tstat}
        \hat t_{0,s}^\star
        =
        \sup_{P \in \mathcal{P}_s} \hat t_0^{(P)},
        \qquad
        \hat t_0^{(P)}
        =
        \sqrt{n - \operatorname{rank}(P) - 1}\;
        \frac{x_0^\top (I - P)\, y}
        {\sqrt{x_0^\top (I - P)\, x_0}\,\sqrt{y^\top (I - P)\, y}},
\end{equation}
is not reducible to the searched coefficient, where the coordinate model
passed from one to the other by the one-line comparison of
Section~\ref{sec:tstat}. Studentizing adds the residual outcome energy
$y^\top (I - P)\, y$ to the denominator, so that the projector acts on
three quadratic forms rather than two and the optimizer trades the
numerator against both denominators. This leads to a more complicated fixed-point
equation, which nonetheless yields some insight.}

\tc{To describe it, some notation needs to be introduced, the covariance
of the reported pair and the part of it the control span reaches,
\begin{equation}\label{eq:cont-gram}
        \Omega
        =
        \begin{pmatrix}
        \sigma_0^2 & \sigma_{0y} \\ \sigma_{0y} & \sigma_y^2
        \end{pmatrix},
        \qquad
        \Omega_\kappa
        =
        \begin{pmatrix}
        A & A_{0y} \\ A_{0y} & A_y
        \end{pmatrix},
        \qquad
        \sigma_{0y} = \Cov(x_0^{(i)}, y^{(i)})
        = \beta_0 \sigma_0^2 + d^\top \beta_{-0},
\end{equation}
with $\sigma_y^2$ the outcome variance \eqref{eq:sigma-y}, $A$ the
treatment energy lying inside the control span, and
\begin{equation}\label{eq:cont-gram-span}
        A = d^\top C^{-1} d + \kappa\,(\sigma_0^2 - d^\top C^{-1} d),
        \quad
        A_{0y} = \beta_0 A + d^\top \beta_{-0},
        \quad
        A_y = \beta_0^2 A + 2 \beta_0\, d^\top \beta_{-0}
        + \beta_{-0}^\top C \beta_{-0} + \kappa\, \sigma_\varepsilon^2 .
\end{equation}
Here $\Omega_\kappa$ is the limiting Gram matrix of the projections of
$x_0$ and $y$ onto the control span, and the difference
$\Omega - \Omega_\kappa$, the geometry search cannot touch, is positive
definite exactly when the outcome noise is non-degenerate.}

\begin{theorem}[Studentized basis search]\label{thm:continuous-tstat}
\tc{Let the assumptions of Theorem~\ref{thm:continuous} hold, with in
addition $\sigma_\varepsilon^2 > 0$. Then
\begin{equation}\label{eq:cont-tstat-limit}
        \frac{\hat t_{0,s}^\star}{\sqrt n}
        \;\toP\;
        \tau^\star
        :=
        \max_{\norm{u}_2 = 1}\;
        \frac{\sigma_{0y} - \alpha_u \beta_u}
        {\sqrt{\left(\sigma_0^2 - \alpha_u^2\right)
        \left(\sigma_y^2 - \beta_u^2\right)}},
        \qquad
        \begin{pmatrix} \alpha_u \\ \beta_u \end{pmatrix}
        = \Omega_\kappa^{1/2}\, u,
\end{equation}
a limit that does not depend on the search budget $s$ and satisfies
$\tau^\star < 1$. It is positive precisely when the limit
\eqref{eq:continuous-limit} of the searched coefficient is positive, and
whenever it is, the maximum in \eqref{eq:cont-tstat-limit} is attained,
and the supremum defining $\hat t_{0,s}^\star$ is attained at a rank-one
projector.}
\end{theorem}

\begin{remark}[One manufactured control, at the parametric rate]
\label{rem:cont-tstat-rankone}
\tc{The maximum in \eqref{eq:cont-tstat-limit} has a concrete reading. A
single control built from the span carries one covariance with the
treatment and one with the outcome, and the pairs of covariances it can
realize form exactly the ellipse
$\{\Omega_\kappa^{1/2} u : \norm{u}_2 = 1\}$. Writing that pair
$(\alpha_u, \beta_u)$, the ratio being maximized is the partial
correlation of $x_0$ and $y$ given that control, so that $\tau^\star$ is
the largest partial correlation one manufactured control can produce. The
searched $t$-statistic therefore grows like $\tau^\star \sqrt n$ with
$\tau^\star$ fixed, under the null as much as under an effect, so that
basis search manufactures significance at the parametric rate at any
budget $s \ge 1$.}
\end{remark}
