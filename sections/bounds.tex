\subsection{Setup}
\label{sec:setup}

% - discrete covariate search
% - searched coefficient and searched t-statistic
% - notation for s, p, n

Throughout the section the data are generated by the model
\eqref{eq:model1}, and the object of study is the coordinate search process
\begin{equation}\label{eq:searched-sup}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S},
        \qquad
        \hat \beta_{0, S} = \frac{x_0^\top M_S\, y}{x_0^\top M_S\, x_0}.
\end{equation}
Throughout, the two design parameters of the search problem are gathered
into a single scale,
\begin{equation}\label{eq:isometry-scale}
        \delta_{s,n,p} = \sqrt{\frac{s \log(ep/s)}{n}},
\end{equation}
the order of the restricted isometry constant at sparsity $2s$ of an
i.i.d.\ subgaussian design
\citep{candesTao2005decoding,baraniuk2008simple}. Fixing the size of the
reported specification is a convenience rather
than a restriction: allowing the analyst to report fewer than $s$
controls changes nothing in what follows, for the reason given in
Remark~\ref{rem:menu-convention} at the end of the section.

The following assumptions are shared by the experimental and
the observational regimes.

\begin{assumption}[Data-generating process and regime]\label{ass:dgp}
The rows $(x_0^{(i)}, X_{-0}^{(i)}, \varepsilon^{(i)})$, $i = 1, \dots, n$,
are independent and identically distributed, and the following hold.
\begin{enumerate}
\item[(i)] \emph{Subgaussian design.} The pair $(x_0^{(i)}, X_{-0}^{(i)})$
is centered and jointly subgaussian: there is a constant $K_\psi$ such that,
for every $u \in \R^{p+1}$,
$\|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{\psi_2}
\le K_\psi \|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{L_2}$, with
$\Var(x_0^{(i)}) = \sigma_0^2 > 0$ and $\Cov(X_{-0}^{(i)}) = C$. The noise
$\varepsilon^{(i)}$ is centered, independent of $(x_0^{(i)}, X_{-0}^{(i)})$,
with $\Var(\varepsilon^{(i)}) = \sigma_\varepsilon^2$ and
$\|\varepsilon^{(i)}\|_{\psi_2} \le K_\psi \sigma_\varepsilon$, and
$\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 > 0$.
\item[(ii)] \emph{Critical regime.} The isometry scale
\eqref{eq:isometry-scale} is small, $\delta_{s,n,p} \le \delta_0$, with
$\delta_0$ a small enough constant depending only on $K_\psi$ of (i);
equivalently $n \ge C_0\, s \log(ep/s)$.
\end{enumerate}
\end{assumption}

Condition (i) is the article's sole distributional hypothesis,
everything else in Assumption~\ref{ass:dgp} constraining the covariance
or the sample size rather than the law of the data. Weakening it to
sub-exponential tails would cost a logarithmic factor in the sample
size, the uniform guarantees the proofs rely on being available only
from $n \gtrsim s \log^2(ep/s)$ \citep{adamczak2011rip}.

Condition (ii) asks that scale to be small, which is the point below
which the denominators of the searched estimators stop concentrating,
and it is the same threshold at which the restricted isometry property
sets in. That the threshold of the present problem coincides with this
well-established one is a finding of this work.

\begin{assumption}[Additional conditions for the lower bounds]
\label{ass:lower}
The two conditions below are consumed by the lower bounds alone, and
continue the numbering of Assumption~\ref{ass:dgp}.
\begin{enumerate}
\item[(iii)] \emph{No near-collinearity of the outcome with few controls.}
No $2s$ controls drawn from a region $A \subseteq [p]$ with
$|A| \ge p/2$ explain more than a fixed share of the outcome variance,
\begin{equation*}
        \max_{J \subseteq A,\, |J| \le 2s}\,
        (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J
        \;\le\;
        (1 - c_{\mathrm{ret}})
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right),
\end{equation*}
for a constant $c_{\mathrm{ret}} \in (0, 1)$.
\item[(iv)] \emph{Moderate deviations and a correlation floor.} With $A$
the region of (iii),
\begin{equation*}
        \log(ep/s) \le c_0 \sqrt n,
        \qquad
        \min_{J \subseteq A,\, |J| \le 2s}
        \lambda_{\min}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)
        \ge c_{\min},
\end{equation*}
for a small enough constant $c_0$ depending only on $K_\psi$ and a
constant $c_{\min} > 0$.
\end{enumerate}
\end{assumption}

Condition (iii) asks that the outcome keep variance of its own,
unexplained by the controls, and asks it only of small groups of them,
which is what fixes the scale of the fluctuations search can find.
Condition (iv) keeps the number of controls in the range where
subgaussian tails still behave like Gaussian ones at the extremes, and
again forbids near-duplicates, its floor on the smallest eigenvalue of
the correlation blocks being weaker than the same floor on the
correlation matrix of all the controls at once. Both would follow from a
single bound on the condition number of the population covariance of
$(x_0, X_{-0}, y)$, as would the restricted conditioning met in
Section~\ref{sec:observational-extensions}, and they are stated apart
because treatment, controls and outcome enter the searched ratio
asymmetrically.


\subsection{The Experimental Case}
\label{sec:sparse-search}

% Note item 1:
% - matching upper and lower bounds
% - logarithmic / critical regime
% - characterize the size of the supremum
% - make clear what assumptions yield matching

\begin{assumption}[Experimental regime]\label{ass:experimental}
The treatment is independent of the candidate controls: $x_0^{(i)}$ is
independent of $X_{-0}^{(i)}$, so that
$d = \Cov(x_0^{(i)}, X_{-0}^{(i)}) = 0$.
\end{assumption}

\begin{theorem}[Matching bounds for experimental coordinate search]
\label{thm:experimental-matching}
Define the scale of the largest outcome variance explained by $s$
controls,
\begin{equation}\label{eq:energy-scale}
        \Psi_s
        =
        \sqrt{\max_{|J| \le s}\, (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J}.
\end{equation}
Under Assumptions~\ref{ass:dgp}, \ref{ass:lower}
and~\ref{ass:experimental}, there are constants $C_u \ge c_\ell > 0$ and
$c, C > 0$, depending only on the constants of the assumptions and not
otherwise on the covariance $C$, such that, with probability at least
$1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:experimental-matching}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \;\in\;
        [c_\ell,\, C_u]\;
        \frac{\delta_{s,n,p}}{\sigma_0}
        \sqrt{\Psi_s^2 + \delta_{s,n,p}^2
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right)},
\end{equation}
where the upper end holds under Assumptions~\ref{ass:dgp}(i)--(ii)
and~\ref{ass:experimental} alone.
\end{theorem}

To begin with, the theorem states that the displacement covariate
search can buy depends primarily on $\delta_{s,n,p}$, the square root of
the share of the sample that the search spends. More specifically, at a
fixed budget $s$ that share vanishes with the sample and so does the
displacement, at rate $\log p / n$ when no small group of controls
predicts the outcome and at worst $\sqrt{\log p / n}$ when some do, so
the reported coefficient converges on the effect. That the number of
covariates the analyst may consult enters only through $\log p$ is a
first finding of this work, since what governs the displacement is how
many controls are reported and hardly how many were available. On the
other hand, as $s$ grows to the scale of the sample, the setting in which
a researcher has as many admissible specifications as observations, the
share is of order one and so is the displacement, which then reaches the
scale of $\beta_0$ itself.

The bound depends on two terms,
$\delta_{s,n,p}^2 \sqrt{\beta_{-0}^\top C \beta_{-0}
+ \sigma_\varepsilon^2} / \sigma_0$ and
$\delta_{s,n,p} \Psi_s / \sigma_0$. When data are abundant and the budget
$s$ is small, $\delta_{s,n,p}$ is small and the term in $\Psi_s$ is the
larger of the two. When few controls actually contribute, however,
$\Psi_s$ stops growing once $s$ passes $\norm{\beta_{-0}}_0$, being
pinned at $\sqrt{\beta_{-0}^\top C \beta_{-0}}$ from there on while
$\delta_{s,n,p}$ keeps rising with the budget, so the first term takes
over.

The theorem is to be compared with the scale of a single fixed
configuration,
\begin{equation*}
        \Var\big( \hat\beta_{0,S} \big)
        \;\asymp\;
        \frac{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}
        {n\, \sigma_0^2}.
\end{equation*}
The relative displacement is then worth up to $\sqrt{s \log(ep/s)}$,
which does not decay even at a fixed budget, and while it stays small
against the true effect size $\beta_0$, this will become problematic for
hypothesis testing, the subject of Section~\ref{sec:tstat}.

\subsection{The Observational Case}
\label{sec:observational-extensions}

% Contribution 2, case 2: confounded (d != 0).

Allowing the treatment to correlate with the controls, $d \ne 0$,
changes the problem in ways that are subtle but decisive, and in general
the two ends of the bound no longer match, diagonal population covariance
matrices under additional assumptions being the exception, and the upper
bound acquires a new factor, the restricted condition number
$\kappa_s(C)$ of Assumption~\ref{ass:observational}(ii), which equals one
for diagonal $C$. The searched supremum is now compared with
$\hat\beta_{0,S^\star}$, where $S^\star$ maximizes the population
coefficient and depends on population quantities alone, so that
identifying it is itself a combinatorial problem over the
$\binom{p}{s}$ configurations, which this work does not take up. The
lower bound calls for further arguments still.

\begin{assumption}[Observational regime]\label{ass:observational}
The following two conditions hold.
\begin{enumerate}
\item[(i)] \emph{Overlap.} The controls do not explain the treatment
entirely, in the sense that, for a constant
$c_{\mathrm{ov}} \in (0, 1]$,
\begin{equation*}
        d^\top C^{-1} d \;\le\; (1 - c_{\mathrm{ov}})\, \sigma_0^2 .
\end{equation*}
\item[(ii)] \emph{Restricted conditioning.} The correlation blocks the
search can reach are well conditioned, in the sense that, for a constant
$K \ge 1$,
\begin{equation*}
        \kappa_s(C)
        = \sup_{|J| \le 2s}
        \frac{\lambda_{\max}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)}
        {\lambda_{\min}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)}
        \;\le\; K .
\end{equation*}
\end{enumerate}
\end{assumption}

\begin{theorem}[Upper bound for observational coordinate search]
\label{thm:observational-bounds}
Fix a drift-maximizing configuration
$S^\star \in \arg\max_{|S| = s} \beta^\star_{0,S}$. Under
Assumptions~\ref{ass:dgp} and~\ref{ass:observational}, there are
constants $C_u, c, C > 0$, depending only on the constants of the
assumptions, such
that, with probability at least
$1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:observational-upper}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat \beta_{0, S^\star}
        \;\le\;
        C_u \sqrt{\kappa_s(C)}\;
        \frac{\delta_{s,n,p}}{\sigma_0}
        \sqrt{\Psi_s^2 + \delta_{s,n,p}^2
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right)}.
\end{equation}
\end{theorem}

The restricted condition number constrains the covariance only
through the small blocks search can reach, and since every such block
inherits the spectrum of the full matrix,
$\kappa_s(C) \le \kappa(C) := \lambda_{\max}(C)/\lambda_{\min}(C)$, the
hypothesis is strictly weaker than the bounded condition number
routinely assumed in random-matrix work. Essentially, it prohibits
near-duplicate controls among the covariates, a single pair at
correlation $1-\varepsilon$ already forcing
$\kappa_s(C) \ge (2-\varepsilon)/\varepsilon$ at every budget, since the
blocks range over $2s$ controls.

\tc{The gap between a configuration's population coefficient and the
effect of interest is the omitted-variable drift
$b(S) = \beta^\star_{0,S} - \beta_0$, explicitly
\begin{equation}\label{eq:drift}
        b(S)
        =
        \frac{\beta_{-0,S^c}^\top
        \left( d_{S^c} - C_{S^c S}\, C_{SS}^{-1}\, d_S \right)}
        {\sigma_0^2 - d_S^\top C_{SS}^{-1} d_S}:
\end{equation}
the numerator collects the outcome signal on the omitted controls,
weighted by the part of their confounding that the included controls
leave unexplained, and the denominator is the treatment variance left
after adjustment. How much of the upper bound search can realize is
governed not by the size of the menu but by the entropy of its
near-maximal part: for $\delta \ge 0$,
\begin{equation}\label{eq:landscape-entropy}
        \ell_\delta
        =
        \log \left| \left\{ S \subseteq [p] : |S| = s,\;
        b(S) \ge b(S^\star) - \delta \right\} \right|
\end{equation}
counts the configurations whose drift lies within $\delta$ of the
best.}

\begin{theorem}[Landscape lower bound]
\label{thm:observational-lower}
\tc{Assume, in addition to the hypotheses of
Theorem~\ref{thm:observational-bounds}, Assumption~\ref{ass:lower}.
There are constants
$c_\ell, c, C > 0$, depending only on the constants of the assumptions,
such that, for every $\delta \ge 0$, with probability at least
$1 - C e^{-c\, \ell_\delta} - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:observational-lower}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat \beta_{0, S^\star}
        \;\ge\;
        c_\ell\,
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}
        {\sigma_0}\, \frac{\ell_\delta}{n}
        \;-\; \delta.
\end{equation}}
\end{theorem}

\tc{Centering at the estimator of a drift-maximizing configuration removes
the sampling noise common to all specifications, exactly as
$\hat\beta_{0,\emptyset}$ does in the experimental case. The
population-centered reading follows by adding back
$\hat\beta_{0,S^\star} - \sup_{|S| = s} \beta^\star_{0,S}$, the ordinary
sampling noise of one configuration, of order $1/\sqrt n$: the searched
estimate exceeds the largest population coefficient by the fluctuation
term whenever the latter dominates, and fluctuates around it otherwise.}

\tc{The center of the bounds is itself a searched quantity: where
randomization pins every configuration at $\beta_0$, confounding gives each
set its own population value, and the reported estimate concentrates
around the largest of the $\binom{p}{s}$ population coefficients. This
center does not shrink with $n$: more data tightens the fluctuation
interval but leaves $\sup_S \beta^\star_{0,S}$ untouched, so specification
search in observational designs is a population problem before it is a
sampling one, and the population side inherits the combinatorial structure
of the menu: evaluating $\sup_S \beta^\star_{0,S}$ requires optimizing
over the same $\binom{p}{s}$ sets the analyst searches. This combinatorial
explosion, absent under randomization, is the substantive difference
between the two regimes. At $d = 0$ every $\beta^\star_{0,S} = \beta_0$,
any configuration is drift-maximizing, $\ell_0 = \log\binom{p}{s}
\ge \tfrac{2}{5}\, s\log(ep/s)$ for $s \le p/2$, and the two theorems
recover the order of Theorem~\ref{thm:experimental-matching}, with
$\hat\beta_{0,S^\star}$ in place of $\hat\beta_{0,\emptyset}$ and up to
the factor $\sqrt{\kappa_s(C)}$ on the upper bound, which the
experimental case shows to be removable.}

\begin{remark}[The drift landscape]
\label{rem:obs-obstruction}
\tc{The lower bound is landscape-adaptive: the entropy $\ell_\delta$ of
the near-maximal family replaces the entropy $s\log(ep/s)$ of the full
menu, and $\delta$ trades resolution against richness. On flat
landscapes --- whenever no single control carries a decisive share of
the confounding, so that one-control swaps move $b$ by less than the
fluctuation scale --- taking $\delta$ a small multiple of
$\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}\,
s\log(ep/s)/(\sigma_0 n)$ gives $\ell_\delta$ of order
$s\log(ep/s)$ and the two bounds match up to $\sqrt{\kappa_s(C)}$. Some
landscape dependence is unavoidable: with $d \ne 0$ and general $C$ the
family $\{ b(S) \}$ can encode hard combinatorial structure, and when a
single dominant confounder separates the drifts by more than the
fluctuation scale, every useful resolution leaves $\ell_\delta$ of
order one --- search cannot combine a near-extremal drift with a
near-extremal fluctuation, and how much of the upper bound is attained
is a property of the landscape, not of the sample. Whether the factor
$\sqrt{\kappa_s(C)}$ is attained under confounding is open: search now
moves the residualized treatment itself, gaining access to the
conditioning of $C$ through the overlap quantities
$d_S^\top C_{SS}^{-1} d_S$.}
\end{remark}

\subsection{The t-Statistic and Consequences for Replicability}
\label{sec:tstat}

\tc{That searching specifications inflates nominal significance is a
classical observation: regressors selected on the very data used for
testing bias fit and test statistics upward, and screened noise covariates
routinely deliver significant $t$- and $F$-statistics
\citep{rencherPun1980inflation, lovell1983data, freedman1983screening,
denton1985industry}. The data-snooping corrections of
\citet{white2000reality} and \citet{romanoWolf2005stepwise} answer by
calibrating tests on the distribution of the maximal statistic over the
searched family.}

\tc{The object here is that maximal statistic itself, the
searched t-statistic
\begin{equation}\label{eq:tstat-def}
        \hat t_{0,S}
        =
        \frac{\hat\beta_{0,S}}{\hat\sigma_S / \sqrt{x_0^\top M_S\, x_0}}
        =
        \sqrt{n-s-1}\,
        \frac{x_0^\top M_S\, y}
        {\sqrt{x_0^\top M_S\, x_0}\, \sqrt{y^\top M_S\, y}},
        \qquad
        \hat\sigma_S^2 = \frac{y^\top M_S\, y}{n-s-1}.
\end{equation}
In the coordinate model, the complexity of searched significance is
contained in the geometry of the raw coefficient: the searched t-statistic
follows from the searched suprema of Sections~\ref{sec:sparse-search}
and~\ref{sec:observational-extensions} by a one-line comparison, and no
structure of $C$ enters the passage beyond two variance floors. This is a
statement about coordinate selection only; for the continuous search of
Section~\ref{sec:continuous-relaxation}, studentization changes the
variational problem itself and no such reduction holds. The mechanism sits
in the denominator: relative to $\hat\beta_{0,S}$, half a power of
$x_0^\top M_S x_0$ is removed and half a power of $y^\top M_S y$ is added.
The added half-power closes one hacking channel: omitting a contributing
covariate ($\beta_j \neq 0$) inflates the residual outcome energy
$y^\top M_S y$ and pushes significance further away, while the denominator
of the raw coefficient never sees the outcome.}

\tc{The comparison requires the two studentizing energies to be
non-degenerate. The population outcome variance and the outcome analogue
of the overlap quantities,
\begin{equation}\label{eq:sigma-y}
        \sigma_y^2
        =
        \beta_0^2 \sigma_0^2
        + 2 \beta_0\, d^\top \beta_{-0}
        + \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2,
        \qquad
        c_{y,S}
        =
        \beta_0\, d_S + (C \beta_{-0})_S,
\end{equation}
describe them at the population level:
$c_{y,S}^\top C_{SS}^{-1} c_{y,S}$ is the outcome variance explained by
the controls in $S$.}

\begin{assumption}[Outcome floor]\label{ass:outcome-floor}
\tc{There is a constant $c_y \in (0, 1]$ such that
\begin{equation*}
        \sigma_y^2
        - \sup_{S \subseteq [p],\, |S| = s}
        c_{y,S}^\top C_{SS}^{-1} c_{y,S}
        \;\ge\;
        c_y\, \sigma_y^2:
\end{equation*}
no $s$ controls explain more than a fixed fraction of the outcome
variance.}
\end{assumption}

\begin{theorem}[Transfer to the searched t-statistic]
\label{thm:tstat-transfer}
\tc{Let Assumptions~\ref{ass:dgp}(i)--(ii), \ref{ass:observational}(i)
and~\ref{ass:outcome-floor} hold (under Assumption~\ref{ass:experimental},
Assumption~\ref{ass:observational}(i) holds with $c_{\mathrm{ov}} = 1$).
There are constants $C_t \ge c_t > 0$ and $c, C > 0$, depending only on
$(K_\psi, c_{\mathrm{ov}}, c_y, C_0)$, such that, with probability at
least $1 - C e^{-c\, s \log(ep/s)}$, on the event
$\sup_{|S| = s} \hat\beta_{0,S} \ge 0$,
\begin{equation}\label{eq:tstat-transfer}
        c_t\, \sqrt{n}\; \frac{\sigma_0}{\sigma_y}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        \;\le\;
        \sup_{S \subseteq [p],\, |S| = s} \hat t_{0, S}
        \;\le\;
        C_t\, \sqrt{n}\; \frac{\sigma_0}{\sigma_y}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}.
\end{equation}}
\end{theorem}

\tc{The searched t-statistic is $\sqrt n$ times the standardized searched
coefficient, up to constants: given the two floors, no further structure of
$C$ enters the comparison, and every bound of
Theorems~\ref{thm:experimental-matching}
and~\ref{thm:observational-bounds} transfers to the t-statistic after
multiplication by $\sqrt n\, \sigma_0/\sigma_y$. In the experimental null
case the transfer applied to Theorem~\ref{thm:experimental-matching} gives
$\sup_{|S|=s} \hat t_{0,S} \asymp s \log(ep/s)/\sqrt n$, the source of the
significance budget $\sqrt n / \log(ep/s)$, a full factor $\sqrt n$ below
the budget $n / \log(ep/s)$ at which search displaces the raw coefficient
by the order of the effect itself: with that many useless
covariates the searched t-statistic reaches any fixed threshold, and at
the critical scaling $n \asymp s \log(ep/s)$ it grows like
$\sqrt{s \log(ep/s)}$.}

\begin{remark}[Reporting fewer than $s$ controls]\label{rem:menu-convention}
\tc{Every result of this section was stated for specifications of exact
size $s$, the convention announced in Section~\ref{sec:setup}. Nothing
changes if the analyst may report any specification with at most $s$
controls. The lower bounds transfer at once, the larger menu containing
the smaller one. The upper bounds transfer because the two menus are of
the same size on the logarithmic scale that governs every bound above:
the sample-size requirement of Assumption~\ref{ass:dgp}(ii) and the rate
$s \log(ep/s) / n$ are untouched, and only the constants move. The same
applies to the searched $t$-statistic.}
\end{remark}
