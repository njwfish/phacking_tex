\subsection{Setup}
\label{sec:setup}

% - discrete covariate search
% - searched coefficient and searched t-statistic
% - notation for s, p, n

Throughout the section the data are generated by the model
\eqref{eq:model1}, and the object of study is the coordinate search process
\begin{equation}\label{eq:searched-sup}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S},
        \qquad
        \hat \beta_{0, S} = \frac{x_0^\top M_S\, y}{x_0^\top M_S\, x_0}.
\end{equation}
Throughout, the two design parameters of the search problem are gathered
into a single scale,
\begin{equation}\label{eq:isometry-scale}
        \delta_{s,n,p} = \sqrt{\frac{s \log(ep/s)}{n}},
\end{equation}
the order of the restricted isometry constant at sparsity $2s$ of an
i.i.d.\ subgaussian design
\citep{candesTao2005decoding,baraniuk2008simple}. Fixing the size of the
reported specification is a convenience rather than a restriction, since
allowing the analyst to report fewer than $s$ controls changes nothing in
what follows, the two families of specifications being of the same size
on the logarithmic scale that governs every bound below.

The following assumptions are shared by the experimental and
the observational regimes.

\begin{assumption}[Data-generating process and regime]\label{ass:dgp}
The rows $(x_0^{(i)}, X_{-0}^{(i)}, \varepsilon^{(i)})$, $i = 1, \dots, n$,
are independent and identically distributed, and the following hold.
\begin{enumerate}
\item[(i)] \emph{Subgaussian design.} The pair $(x_0^{(i)}, X_{-0}^{(i)})$
is centered and jointly subgaussian: there is a constant $K_\psi$ such that,
for every $u \in \R^{p+1}$,
$\|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{\psi_2}
\le K_\psi \|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{L_2}$, with
$\Var(x_0^{(i)}) = \sigma_0^2 > 0$ and $\Cov(X_{-0}^{(i)}) = C$. The noise
$\varepsilon^{(i)}$ is centered, independent of $(x_0^{(i)}, X_{-0}^{(i)})$,
with $\Var(\varepsilon^{(i)}) = \sigma_\varepsilon^2$ and
$\|\varepsilon^{(i)}\|_{\psi_2} \le K_\psi \sigma_\varepsilon$, and
$\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 > 0$.
\item[(ii)] \emph{Critical regime.} The isometry scale
\eqref{eq:isometry-scale} is small, $\delta_{s,n,p} \le \delta_0$, with
$\delta_0$ a small enough constant depending only on $K_\psi$ of (i);
equivalently $n \ge C_0\, s \log(ep/s)$.
\end{enumerate}
\end{assumption}

Condition (i) is the article's sole distributional hypothesis,
everything else in Assumption~\ref{ass:dgp} constraining the covariance
or the sample size rather than the law of the data. Weakening it to
sub-exponential tails would cost a logarithmic factor in the sample
size, the uniform guarantees the proofs rely on being available only
from $n \gtrsim s \log^2(ep/s)$ \citep{adamczak2011rip}.

Condition (ii) asks that scale to be small, which is the point below
which the denominators of the searched estimators stop concentrating,
and it is the same threshold at which the restricted isometry property
sets in. That the threshold of the present problem coincides with this
well-established one is a finding of this work.

\begin{assumption}[Additional conditions for the lower bounds]
\label{ass:lower}
The two conditions below are consumed by the lower bounds alone, and
continue the numbering of Assumption~\ref{ass:dgp}.
\begin{enumerate}
\item[(iii)] \emph{No near-collinearity of the outcome with few controls.}
No $2s$ controls drawn from a region $A \subseteq [p]$ with
$|A| \ge p/2$ explain more than a fixed share of the outcome variance,
\begin{equation*}
        \max_{J \subseteq A,\, |J| \le 2s}\,
        (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J
        \;\le\;
        (1 - c_{\mathrm{ret}})
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right),
\end{equation*}
for a constant $c_{\mathrm{ret}} \in (0, 1)$.
\item[(iv)] \emph{Moderate deviations and a correlation floor.} With $A$
the region of (iii),
\begin{equation*}
        \log(ep/s) \le c_0 \sqrt n,
        \qquad
        \min_{J \subseteq A,\, |J| \le 2s}
        \lambda_{\min}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)
        \ge c_{\min},
\end{equation*}
for a small enough constant $c_0$ depending only on $K_\psi$ and a
constant $c_{\min} > 0$.
\end{enumerate}
\end{assumption}

Condition (iii) asks that the outcome keep variance of its own,
unexplained by the controls, and asks it only of small groups of them,
which is what fixes the scale of the fluctuations search can find.
Condition (iv) keeps the number of controls in the range where
subgaussian tails still behave like Gaussian ones at the extremes, and
again forbids near-duplicates, its floor on the smallest eigenvalue of
the correlation blocks being weaker than the same floor on the
correlation matrix of all the controls at once. Both would follow from a
single bound on the condition number of the population covariance of
$(x_0, X_{-0}, y)$, as would the restricted conditioning met in
Section~\ref{sec:observational-extensions}, and they are stated apart
because treatment, controls and outcome enter the searched ratio
asymmetrically.


\subsection{The Experimental Case}
\label{sec:sparse-search}

% Note item 1:
% - matching upper and lower bounds
% - logarithmic / critical regime
% - characterize the size of the supremum
% - make clear what assumptions yield matching

\begin{assumption}[Experimental regime]\label{ass:experimental}
The treatment is independent of the candidate controls: $x_0^{(i)}$ is
independent of $X_{-0}^{(i)}$, so that
$d = \Cov(x_0^{(i)}, X_{-0}^{(i)}) = 0$.
\end{assumption}

\begin{theorem}[Matching bounds for experimental coordinate search]
\label{thm:experimental-matching}
Define the scale of the largest outcome variance explained by $s$
controls,
\begin{equation}\label{eq:energy-scale}
        \Psi_s
        =
        \sqrt{\max_{|J| \le s}\, (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J}.
\end{equation}
Under Assumptions~\ref{ass:dgp}, \ref{ass:lower}
and~\ref{ass:experimental}, there are constants $C_u \ge c_\ell > 0$ and
$c, C > 0$, depending only on the constants of the assumptions and not
otherwise on the covariance $C$, such that, with probability at least
$1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:experimental-matching}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \;\in\;
        [c_\ell,\, C_u]\;
        \frac{\delta_{s,n,p}}{\sigma_0}
        \sqrt{\Psi_s^2 + \delta_{s,n,p}^2
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right)},
\end{equation}
where the upper end holds under Assumptions~\ref{ass:dgp}(i)--(ii)
and~\ref{ass:experimental} alone.
\end{theorem}

To begin with, the theorem states that the displacement covariate
search can buy depends primarily on $\delta_{s,n,p}$, the square root of
the share of the sample that the search spends. More specifically, at a
fixed budget $s$ that share vanishes with the sample and so does the
displacement, at rate $\log p / n$ when no small group of controls
predicts the outcome and at worst $\sqrt{\log p / n}$ when some do, so
the reported coefficient converges on the effect. That the number of
covariates the analyst may consult enters only through $\log p$ is a
first finding of this work, since what governs the displacement is how
many controls are reported and hardly how many were available. On the
other hand, as $s$ grows to the scale of the sample, the setting in which
a researcher has as many admissible specifications as observations, the
share is of order one and so is the displacement, which then reaches the
scale of $\beta_0$ itself.

The bound depends on two terms,
$\delta_{s,n,p}^2 \sqrt{\beta_{-0}^\top C \beta_{-0}
+ \sigma_\varepsilon^2} / \sigma_0$ and
$\delta_{s,n,p} \Psi_s / \sigma_0$. Writing
\begin{equation}\label{eq:explained-share}
        \theta_s
        =
        \frac{\Psi_s^2}
        {\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}
\end{equation}
for the share of the outcome variance, net of the treatment, that the
best $s$ controls explain in population,
the second term is the larger exactly when
$\theta_s \ge \delta_{s,n,p}^2$, which is the case when data are abundant
and the budget $s$ is small. Since the best $s$ controls explain no more than all of
them, $\Psi_s^2 \le \beta_{-0}^\top C \beta_{-0}$ at every budget, and
when few controls actually contribute $\Psi_s$ stops growing once $s$
passes $\norm{\beta_{-0}}_0$, being pinned at that value from there on
while $\delta_{s,n,p}$ keeps rising with the budget, so the first term
takes over.

Figure~\ref{fig:randomized-coordinate-geometry} is designed to establish a
clear interpretation of $\theta_s$ in a simple simulation setting. We
randomize the treatment $x_0^{(i)} \sim \operatorname{Bern}(1/2)$, fixing $s$ and
$\sigma_\varepsilon^2$, and vary the number $p$ of available controls
$X_{-0}^{(i)} \sim \mathcal{N}(0,I_p)$ and the sample size $n$. The treatment
has no effect on the outcome, $\beta_0=0$. The key additional idea is the number
$k$ of ``prognostic controls'' that actually carry signal for the outcome.

The $k$ nonzero coordinates of $\beta_{-0}$ have equal magnitude and are
rescaled as $k$ varies so that $\norm{\beta_{-0}}_2^2$ remains fixed. For
$k>0$, this gives
\begin{equation*}
        \theta_s
        =
        \frac{\norm{\beta_{-0}}_2^2}
        {\norm{\beta_{-0}}_2^2+\sigma_\varepsilon^2}
        \min\left\{1,\frac{s}{k}\right\},
\end{equation*}
while $k=0$ gives $\theta_s=0$. Increasing $p$ at fixed $k$ changes
$\delta_{s,n,p}$ without changing $\theta_s$; increasing $k$ at fixed $p$
spreads the same outcome signal over more controls and lowers $\theta_s$;
increasing $n$ lowers $\delta_{s,n,p}$ without changing $\theta_s$.

For each design, we begin with the no-control coefficient
$\hat\beta_{0,\emptyset}$ and add the control that produces the largest current
value of $|\hat\beta_{0,S}|$ until exactly $s$ controls have been selected. The
figure reports the resulting mean increase in coefficient magnitude. This
forward search is an attainable counterpart to the exhaustive coordinate
maximum in \eqref{eq:searched-sup}. The rate in
\eqref{eq:experimental-matching} predicts a displacement of order
$\delta_{s,n,p}^2$ when $\theta_s=0$ and of order
$\delta_{s,n,p}\sqrt{\theta_s}$ when the prognostic term dominates. The panels
show these two regimes directly: enlarging the menu raises the displacement,
concentrating a fixed prognostic signal raises it further, and increasing $n$
drives both coefficient displacements toward zero at their respective rates.

Figure~\ref{fig:covariance-prognostic-geometry} asks how covariance changes the
same prognostic structure. We now fix $n$, $p$, $s$, $k$, and the total outcome
variance, but replace $I_p$ with a covariance matrix containing a rank-one
component of strength $\rho$. For each value of $\rho$, the covariance spectrum
is fixed while its leading direction is rotated. At alignment $a=0$, this
direction is supported on the $k$ prognostic controls; at $a=1$, it is spread
equally across all $p$ controls.

This rotation leaves the total prognostic signal and the eigenvalues of $C$
unchanged, but it changes $C\beta_{-0}$ and therefore $\Psi_s$ and $\theta_s$.
The left panel summarizes the redistribution by the participation ratio of the
squared marginal outcome correlations, which is the effective number of
prognostic controls. The right panel reports the corresponding coefficient
gain. As $a$ increases, correlation first creates additional prognostic proxies
and then makes those proxies increasingly redundant, producing the observed
nonmonotonicity. Covariance therefore matters through the sparse outcome
variation it makes accessible, not through the overall strength of dependence
alone: designs with the same covariance eigenvalues can have different values
of $\theta_s$ and different susceptibility to coordinate search.

The theorem is to be compared with the scale of a single fixed
configuration,
\begin{equation*}
        \Var\big( \hat\beta_{0,S} \big)
        \;\asymp\;
        \frac{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}
        {n\, \sigma_0^2}.
\end{equation*}
The relative displacement is then
$\sqrt{s \log(ep/s)}\, \sqrt{\theta_s + \delta_{s,n,p}^2}$, at most
$\sqrt{s \log(ep/s)}$, and it does not decay at a fixed budget. So while
the displacement stays small against the true effect size $\beta_0$, this
will become problematic for hypothesis testing, the subject of
Section~\ref{sec:tstat}.

\subsection{The Observational Case}
\label{sec:observational-extensions}

% Contribution 2, case 2: confounded (d != 0).

Allowing the treatment to correlate with the controls, $d \ne 0$,
changes the problem in one important aspect, the population coefficients
$\beta^\star_{0,S}$ no longer agreeing across configurations. The searched
supremum needs now to be compared with $\hat\beta_{0,S^\star}$
($S^\star \in \arg\max_{|S| = s} \beta^\star_{0,S}$), which splits the
problem in two, the choice of the most biased specification and what
search adds on top of it, and without further assumptions identifying
$S^\star$ may be a combinatorial problem over the $\binom{p}{s}$
configurations, out of the scope of this work. The upper bound also
acquires a new factor, the restricted condition number $\kappa_s(C)$ of
Assumption~\ref{ass:observational}(ii), and the two ends no longer match
in general.

\begin{assumption}[Observational regime]\label{ass:observational}
The following two conditions hold.
\begin{enumerate}
\item[(i)] \emph{Overlap.} The controls do not explain the treatment
entirely, in the sense that, for a constant
$c_{\mathrm{ov}} \in (0, 1]$,
\begin{equation*}
        d^\top C^{-1} d \;\le\; (1 - c_{\mathrm{ov}})\, \sigma_0^2 .
\end{equation*}
\item[(ii)] \emph{Restricted conditioning.} The correlation blocks the
search can reach are well conditioned, in the sense that, for a constant
$K \ge 1$,
\begin{equation*}
        \kappa_s(C)
        = \sup_{|J| \le 2s}
        \frac{\lambda_{\max}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)}
        {\lambda_{\min}\!\left( \diag(C_{JJ})^{-1/2}\, C_{JJ}\,
        \diag(C_{JJ})^{-1/2} \right)}
        \;\le\; K .
\end{equation*}
\end{enumerate}
\end{assumption}

The restricted condition number constrains the covariance only
through the small blocks search can reach, and since every such block
inherits the spectrum of the full matrix,
$\kappa_s(C) \le \kappa(C) := \lambda_{\max}(C)/\lambda_{\min}(C)$, the
hypothesis is weaker than the bounded condition number routinely assumed
in random-matrix work.

\begin{theorem}[Upper bound for observational coordinate search]
\label{thm:observational-bounds}
Let $S^\star \in \arg\max_{|S| = s} \beta^\star_{0,S}$. Under
Assumptions~\ref{ass:dgp} and~\ref{ass:observational}, there are
constants $C_u, c, C > 0$, depending only on the constants of the
assumptions, such that, with probability at least
$1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:observational-upper}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat \beta_{0, S^\star}
        \;\le\;
        C_u \sqrt{\kappa_s(C)}\;
        \frac{\delta_{s,n,p}}{\sigma_0}
        \sqrt{\Psi_s^2 + \delta_{s,n,p}^2
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right)}.
\end{equation}
\end{theorem}

Two features distinguish this statement from
Theorem~\ref{thm:experimental-matching}, the factor
$\sqrt{\kappa_s(C)}$, which $d = 0$ removes, and the centering at
$\hat\beta_{0,S^\star}$, which leaves the stochastic part alone on the
left.

Turning to lower bounds, it is now required to introduce the population
process $\{ \beta^\star_{0,S} \}_{|S| = s}$, what the searched process
becomes once the sampling randomness is removed, explicitly defined as
\begin{equation}\label{eq:drift}
        \beta^\star_{0,S} = \beta_0 + b(S),
        \qquad
        b(S)
        =
        \frac{\beta_{-0,S^c}^\top
        \left( d_{S^c} - C_{S^c S}\, C_{SS}^{-1}\, d_S \right)}
        {\sigma_0^2 - d_S^\top C_{SS}^{-1} d_S}.
\end{equation}
We may define its entropy, the number of configurations whose population
coefficient lies within a margin $\delta \ge 0$ of the largest, as
\begin{equation}\label{eq:landscape-entropy}
        \ell_\delta
        =
        \log \left| \left\{ S \subseteq [p] : |S| = s,\;
        \beta^\star_{0,S} \ge \beta^\star_{0,S^\star} - \delta
        \right\} \right| .
\end{equation}

\begin{theorem}[Lower bound for observational coordinate search]
\label{thm:observational-lower}
Assume, in addition to the hypotheses of
Theorem~\ref{thm:observational-bounds}, Assumption~\ref{ass:lower}.
There are constants $c_\ell, c, C > 0$, depending only on the constants
of the assumptions, such that, for every $\delta \ge 0$, with
probability at least
$1 - C e^{-c\, \ell_\delta} - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:observational-lower}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat \beta_{0, S^\star}
        \;\ge\;
        c_\ell\,
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}
        {\sigma_0}\, \frac{\ell_\delta}{n}
        \;-\; \delta.
\end{equation}
\end{theorem}

Whether upper and lower bounds match, even having ruled out the
$\kappa_s(C)$ factor by setting $C = I$, is decided by $\ell_\delta$. On
the one hand, uniform loadings $d_j = \rho\, \sigma_0 / \sqrt p$ and
$\beta_{-0,j} = b / \sqrt p$ give every configuration the same drift, so
that $\ell_\delta = \log \binom{p}{s} \ge \tfrac{2}{5}\, s \log(ep/s)$
(for $s \le p/2$) at every margin and, in the critical regime, both ends
are of order
$\delta_{s,n,p}^2 \sqrt{b^2 + \sigma_\varepsilon^2} / \sigma_0$. On the
other hand, when the supremum of the population process is well
separated, as when the same design is perturbed to distinct loadings and
the maximizer, which keeps the $s$ smallest of them, leads its runner-up
by $\rho\, \sigma_0 (\beta_{-0,(s+1)} - \beta_{-0,(s)})
/ (\sqrt p\, \sigma_0^2)$, a gap above the fluctuation scale leaves
$\ell_\delta = 0$. In this latter case the potential for hacking through
stochastic effects is lower, and more accurately described by the lower
bound.

Figure~\ref{fig:observational-coordinate-geometry} introduces the population
drift that arises when $d \ne 0$. The treatment remains Bernoulli, but its
assignment probability depends on a score constructed from the controls. We
index the strength of this dependence by
$R_T^2=d^\top C^{-1}d/\sigma_0^2$, the population share of treatment variance
explained by the controls, and vary $R_T^2$ together with $n$. The treatment
score is orthogonal to the prognostic score, so the no-control coefficient has
no population omitted-variable bias.

Specification search need not preserve this orthogonality. A balanced set of
controls does, while an unbalanced set creates a nonzero drift $b(S)$. At the
randomized endpoint $R_T^2=0$, every specification has the same
population coefficient and the searched displacement vanishes with $n$. For
any fixed positive $R_T^2$, the process $\{\beta_{0,S}^\star\}$ has different
centers across supports, and the searched coefficient converges toward the
largest of them. The nonzero limit approached by each confounded curve is
therefore the population drift component of the observational bound.

\subsection{The t-Statistic and Consequences for Replicability}
\label{sec:tstat}

This section covers the behaviour of the $t$-statistic in a testing
problem, and through it the extent to which researcher degrees of freedom
bear on statistical significance. Compared to raw estimation,
substituting one set of controls for another when focusing on the
$t$-statistic amounts to testing related hypotheses with different
estimands, the difficulty being that a reader is shown only the null and
the alternative that ended up being compared. We work under a global null
hypothesis, stated below, with an analyst who computes a $t$-statistic
for each set of controls considered, and study how much that statistic
may be inflated by the comparison of configurations. Instead of the raw
coefficient, substituting controls now modifies two quantities, the
numerator and the residual sum of squares that studentizes it,
\begin{equation}\label{eq:tstat-def}
        \hat t_{0,S}
        =
        \frac{\hat\beta_{0,S}}{\hat\sigma_S / \sqrt{x_0^\top M_S\, x_0}}
        =
        \sqrt{n-s-1}\,
        \frac{x_0^\top M_S\, y}
        {\sqrt{x_0^\top M_S\, x_0}\, \sqrt{y^\top M_S\, y}},
        \qquad
        \hat\sigma_S^2 = \frac{y^\top M_S\, y}{n-s-1},
\end{equation}
where $\hat\sigma_S^2$ is formed from the residuals of $y$ on $X_S$
alone, so that $\hat t_{0,S} = \sqrt{n-s-1}\, \hat\rho_S$ with
$\hat\rho_S$ the sample partial correlation of $x_0$ and $y$ given $X_S$,
the least-squares $t$-statistic being
$\hat t_{0,S} / \sqrt{1 - \hat\rho_S^2}$, an increasing function of it
with the same maximizer.

\begin{assumption}[Testing regime]\label{ass:testing}
The effect is null, $\beta_0 = 0$, and the following two conditions
hold.
\begin{enumerate}
\item[(i)] \emph{Local alternatives.} No admissible specification is
significant on its own, in the sense that, for a constant $K_\rho$,
\begin{equation*}
        \sqrt n \max_{|S| \le s}
        \left| \Corr\big( x_0^{(i)}, y^{(i)} \mid X_S^{(i)} \big) \right|
        \;\le\; K_\rho .
\end{equation*}
\item[(ii)] \emph{Outcome floor.} The explained share
\eqref{eq:explained-share} stays away from one, $\theta_s \le 1 - c_y$
for a constant $c_y \in (0, 1]$.
\end{enumerate}
\end{assumption}

The effect being null, what a specification actually tests is
$\rho(S) = \Corr(x_0^{(i)}, y^{(i)} \mid X_S^{(i)}) = 0$, which the null
on $\beta_0$ leaves open as soon as the treatment is confounded, and
condition (i) admits such alternatives at the local scale $n^{-1/2}$
alone (as local asymptotic analyses do for a single configuration
\citep{vaart1998asymptotic}, keeping the signal comparable with the noise
of that configuration). Here, imposed across the population process, the
same condition keeps the displacement search reaches at that same size,
so that signal, noise and search are commensurable and the power of a
searched test remains a meaningful quantity. The condition holds as soon
as the confounding is of bounded order on that scale, studentization
leaving $\rho(S)^2 \le d^\top C^{-1} d / \sigma_0^2$, so that
$n\, d^\top C^{-1} d = O(\sigma_0^2)$ suffices, and randomization gives
it with $d = 0$ and $\rho(S) = 0$ for every $S$. Condition (ii) asks the
best $s$ controls not to explain the outcome entirely, so that the
residual sum of squares $y^\top M_S y$ of \eqref{eq:tstat-def} keeps a
share $c_y$ of the outcome variance; it is implied by
$\Psi_s^2 \le \beta_{-0}^\top C \beta_{-0}$, which leaves the noise share
$\sigma_\varepsilon^2 / (\beta_{-0}^\top C \beta_{-0}
+ \sigma_\varepsilon^2)$ as an admissible $c_y$.

\begin{theorem}[Searched significance under the null]\label{thm:tstat}
Under Assumptions~\ref{ass:dgp}, \ref{ass:lower}
and~\ref{ass:testing}, there are constants $C_t \ge c_t > 0$ and
$c, C > 0$, depending only on the constants of the assumptions, such
that, with probability at least $1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:tstat-rate}
        \sup_{S \subseteq [p],\, |S| = s} \hat t_{0,S}
        - \hat t_{0,\emptyset}
        \;\in\;
        [c_t,\, C_t]\;
        \sqrt{ s \log(ep/s) }\,
        \sqrt{ \theta_s + \delta_{s,n,p}^2 }
        \;+\; [-C,\, C] .
\end{equation}
\end{theorem}

The radical adds two coefficients of determination. The isometry
scale $\delta_{s,n,p}^2 = s\log(ep/s)/n$ is the one search manufactures,
the largest in-sample fit reached by the best of the $\binom{p}{s}$
configurations when the controls explain nothing, and the explained share
$\theta_s$ of \eqref{eq:explained-share} is the one the controls
genuinely have. Significance is bought either way, and the test breaks as
soon as either exceeds $1/(s\log(ep/s))$,
\begin{equation*}
        s \log(ep/s)
        \;\gtrsim\;
        \min\left\{ \sqrt n,\; \theta_s^{-1} \right\} .
\end{equation*}

Which of the two sets the budget is decided by $\theta_s$ against
$n^{-1/2}$, the manufactured fit taking over below that point, and for
uncorrelated controls of unit variance $\theta_s$ is the share of
$\norm{\beta_{-0}}_2^2 + \sigma_\varepsilon^2$ carried by the $s$ largest
coordinates of $\beta_{-0}$ in absolute value. Where few controls contribute,
$\theta_s$ is pinned at $\norm{\beta_{-0}}_2^2
/ (\norm{\beta_{-0}}_2^2 + \sigma_\varepsilon^2)$ from
$s \ge \norm{\beta_{-0}}_0$ on and the test breaks at
$s\log(ep/s) \gtrsim 1 + \sigma_\varepsilon^2 / \norm{\beta_{-0}}_2^2$, a
threshold set by the outcome's signal-to-noise ratio and not by the
sample size, provided it both covers
$\norm{\beta_{-0}}_0 \log(ep/\norm{\beta_{-0}}_0)$ and stays below
$\sqrt n$.
Coordinates of equal magnitude spread over a support larger than the
budget give $\theta_s$ that same share times $s / \norm{\beta_{-0}}_0$,
which passes below $n^{-1/2}$ once $\norm{\beta_{-0}}_0$ exceeds
$s \sqrt n$ times that share, itself a demand on how many controls there
are to choose from, and there the $\sqrt n$ budget binds instead.
