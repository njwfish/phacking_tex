\subsection{Setup}
\label{sec:setup}

% - discrete covariate search
% - searched coefficient and searched t-statistic
% - notation for s, p, n

\tc{Throughout the section the data are generated by the model
\eqref{eq:model1}, and the object of study is the coordinate search process
\begin{equation}\label{eq:searched-sup}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S},
        \qquad
        \hat \beta_{0, S} = \frac{x_0^\top M_S\, y}{x_0^\top M_S\, x_0}.
\end{equation}
The following assumptions are shared by the experimental and the
observational regimes.}

\begin{assumption}[Data-generating process and regime]\label{ass:dgp}
\tc{The rows $(x_0^{(i)}, X_{-0}^{(i)}, \varepsilon^{(i)})$, $i = 1, \dots, n$,
are independent and identically distributed, and the following hold.
\begin{enumerate}
\item[(i)] \emph{Subgaussian design.} The pair $(x_0^{(i)}, X_{-0}^{(i)})$
is centered and jointly subgaussian: there is a constant $K_\psi$ such that,
for every $u \in \R^{p+1}$,
$\|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{\psi_2}
\le K_\psi \|\langle (x_0^{(i)}, X_{-0}^{(i)}), u\rangle\|_{L_2}$, with
$\Var(x_0^{(i)}) = \sigma_0^2 > 0$ and $\Cov(X_{-0}^{(i)}) = C$. The noise
$\varepsilon^{(i)}$ is centered, independent of $(x_0^{(i)}, X_{-0}^{(i)})$,
with $\Var(\varepsilon^{(i)}) = \sigma_\varepsilon^2$ and
$\|\varepsilon^{(i)}\|_{\psi_2} \le K_\psi \sigma_\varepsilon$, and
$\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 > 0$.
\item[(ii)] \emph{Critical regime.} The sample size satisfies
$n \ge C_0\, s \log(ep/s)$, with $C_0$ a large enough constant depending
only on $K_\psi$ of (i).
\item[(iii)] \emph{Restricted conditioning.} There is a constant $K \ge 1$
such that
\begin{equation*}
        \kappa_s(C)
        = \sup_{|J| \le 2s}
        \frac{\lambda_{\max}(C_{JJ})}{\lambda_{\min}(C_{JJ})}
        \le K.
\end{equation*}
\item[(iv)] \emph{Restricted signal retention.} There are a constant
$c_{\mathrm{ret}} > 0$ and a set $A \subseteq [p]$ with $|A| \ge p/2$ such
that, for every $J \subseteq A$ with $|J| \le 2s$,
\begin{equation*}
        \beta_{-0}^\top C \beta_{-0}
        - (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J
        + \sigma_\varepsilon^2
        \;\ge\;
        c_{\mathrm{ret}}
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right).
\end{equation*}
\item[(v)] \emph{No sparse outcome proxy.} There is a constant
$C_{\mathrm{rep}} \ge 1$ such that
\begin{equation*}
        n \max_{|J| \le s}\,
        (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J
        \;\le\;
        C_{\mathrm{rep}}\; s \log(ep/s)
        \left( \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2 \right).
\end{equation*}
\end{enumerate}}
\end{assumption}

\begin{remark}[Common covariance classes]\label{rem:kappa-classes}
\tc{Eigenvalue interlacing gives
$\kappa_s(C) \le \kappa(C) := \lambda_{\max}(C)/\lambda_{\min}(C)$, so
condition (iii) of Assumption~\ref{ass:dgp} is strictly weaker than a bound on the global
condition number, and the gap can be of order $p/s$: for diagonal $C$ the
restricted quantity equals the variance spread $\max_j c_j / \min_j c_j$,
while for a factor structure $C = I_p + \sum_{k=1}^m \theta_k v_k v_k^\top$
with delocalized directions ($\mu := p \max_k \|v_k\|_\infty^2$ bounded),
$\kappa_s(C) \le 1 + 2 s m \mu\, \theta_{\max}/p$ stays bounded for
$\theta_{\max}$ as large as order $p/(s m \mu)$, even though
$\kappa(C) = 1 + \theta_{\max}$ then diverges. For stationary designs
$C_{ij} = r(|i-j|)$ with spectral density $f$, interlacing gives
$\kappa_s(C) \le \sup f / \inf f$ uniformly in $s$, and the binding sets are
contiguous runs of covariates: under AR(1) dependence $r(k) = \rho^{|k|}$,
adjacent pairs alone force $\kappa_s(C) \ge (1+\rho)/(1-\rho)$, so the
assumption degrades as $\rho \to 1$, when neighboring covariates become
near-duplicates. Local clusters are the sharpest failure: $m$ equicorrelated
covariates at correlation $\rho$ give
$\kappa_s(C) = \left(1 + (\min(2s,m)-1)\rho\right)/(1-\rho)$, and a single
pair at correlation $1 - \varepsilon$ already forces
$\kappa_s(C) \ge (2-\varepsilon)/\varepsilon$ for every $s \ge 1$. What
condition (iii) retains from the global condition number is thus
precisely a no-near-duplicates condition at group size $2s$: global spectral
spread is forgiven, and for unit-diagonal $C$ the assumption is equivalent,
up to a factor $2s$, to the restricted minimal eigenvalue
$\min_{|J| \le 2s} \lambda_{\min}(C_{JJ})$ being bounded away from zero.}
\end{remark}

\tc{The left-hand side of the display in (iv) is
$\Var\!\left(X_{-0}^\top \beta_{-0} \mid X_{-0, J}\right) + \sigma_\varepsilon^2$:
the condition states that no group of at most $2s$ controls drawn from $A$ can
absorb the part of the outcome variance that search exploits. It holds with
$c_{\mathrm{ret}} = 1/2$ and $A$ the $p/2$ coordinates carrying the least
signal whenever $C = I$, and it holds whenever the noise variance
$\sigma_\varepsilon^2$ is at least a constant fraction of
$\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2$. Only the lower bound of
Theorem~\ref{thm:experimental-matching} consumes it. Condition (v) is its
mirror image for the upper bound: the quantity
$(C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J$ is the outcome variance
explained by the controls in $J$, and (v) asks that no $s$ controls explain
the outcome beyond a fixed multiple of the $s\log(ep/s)/n$ estimation floor.
Only the upper bound consumes it.}

\subsection{The Experimental Case}
\label{sec:sparse-search}

% Note item 1:
% - matching upper and lower bounds
% - logarithmic / critical regime
% - characterize the size of the supremum
% - make clear what assumptions yield matching

\tc{The experimental case is the benchmark in which the treatment carries no
population correlation with any candidate control, as guaranteed by
randomization.}

\begin{assumption}[Experimental regime]\label{ass:experimental}
\tc{The treatment is independent of the candidate controls: $x_0^{(i)}$ is
independent of $X_{-0}^{(i)}$, so that
$d = \Cov(x_0^{(i)}, X_{-0}^{(i)}) = 0$.}
\end{assumption}

\begin{theorem}[Matching bounds for experimental coordinate search]
\label{thm:experimental-matching}
\tc{Let Assumptions~\ref{ass:dgp}(i), (ii), (v) and~\ref{ass:experimental}
hold, and write $\hat\beta_{0,\emptyset} = x_0^\top y / x_0^\top x_0$ for the
estimator with no controls. There are constants $C_u, c, C > 0$, depending
only on $(K_\psi, C_{\mathrm{rep}}, C_0)$, such that, with probability at
least $1 - C e^{-c\, s \log(ep/s)}$,
\begin{equation}\label{eq:experimental-upper}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \;\le\;
        C_u\,
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}{\sigma_0}\,
        \frac{s \log(ep/s)}{n}.
\end{equation}
Assume in addition the retention condition (iv) of Assumption~\ref{ass:dgp},
the correlation floor
$\min_{J \subseteq A,\, |J| \le 2s} \lambda_{\min}(R_{JJ}) \ge c_{\min}$ on
the retention region $A$, with $R$ the correlation matrix of $C$, and that
$\log(ep/s) \le c_0 \sqrt{n}$ for a small enough constant $c_0$ depending
only on $K_\psi$. Then there are constants $c_\ell, c', C' > 0$, depending
only on $(K_\psi, c_{\mathrm{ret}}, c_{\min}, C_0)$, such that, with
probability at least $1 - C' e^{-c'\, s \log(ep/s)}$,
\begin{equation}\label{eq:experimental-lower}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \;\ge\;
        c_\ell\,
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}{\sigma_0}\,
        \frac{s \log(ep/s)}{n}.
\end{equation}}
\end{theorem}

\tc{The displacement grows linearly in the search budget $s$ and decays at the
rate $1/n$, while the size $p$ of the control menu enters only through
$\log(ep/s)$: what search extracts is governed by how many controls
enter the reported specification, hardly by how many were available. The
prefactor
$\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2} / \sigma_0$ is the
exploitable energy of the problem, the standard deviation of the outcome
component left unexplained by the treatment per unit of treatment variation;
search feeds on control signal and noise alike: an outcome well explained
by the controls is as hackable as a noisy one whenever no $s$ controls
explain it beyond the estimation floor, in the sense of condition (v).
Randomization pins the
center of the whole ensemble: under $d = 0$ every $\hat\beta_{0,S}$ is
centered at $\beta_0$, and search inflates the spread around the truth without
shifting it, by the amount that Theorem~\ref{thm:experimental-matching} makes
exact. The bounds are centered at the no-control estimator
$\hat\beta_{0,\emptyset}$ because the sampling noise common to every
specification is beyond the reach of search.
Measured against the sampling noise of a single specification, of order
$\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}/(\sigma_0 \sqrt n)$,
the searched displacement represents $s\log(ep/s)/\sqrt n$ standard errors,
exceeding the order of one standard error precisely when
$n \lesssim s^2 \log^2(ep/s)$.}

\begin{corollary}[Fixed budget, growing menu]\label{cor:fixed-s}
\tc{Fix $s \ge 1$ and let $p \to \infty$ with $n \ge C_0\, s \log p$, under
the assumptions of both bounds of Theorem~\ref{thm:experimental-matching}.
With probability at least $1 - C p^{-c}$,
\begin{equation*}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \;\asymp\;
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}{\sigma_0}\,
        \frac{s \log p}{n}.
\end{equation*}
If moreover $\sup_{|J| \le 2s} \|R_{JJ} - I\|_{\mathrm{op}} \to 0$, then,
with $\zeta_1, \dots, \zeta_p$ independent products of two independent
standard Gaussian variables,
\begin{equation*}
        \frac{\sigma_0\, n}
        {\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}
        \left(
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S}
        - \hat\beta_{0,\emptyset}
        \right)
        = \sum_{j=1}^{s} \{\zeta_k\}_{(j)} + o_{\mathbb P}(1),
\end{equation*}
so that the centered maximum is asymptotically Gumbel for $s = 1$, and for
fixed $s \ge 2$ the top-$s$ sum converges to that of a Poisson process with
intensity $e^{-x}\,dx$.}
\end{corollary}

\begin{corollary}[Critical regime]\label{cor:critical}
\tc{Let $s \to \infty$ at the critical scaling
$C_0\, s \log(ep/s) \le n \le C_1\, s \log(ep/s)$, under the assumptions of
both bounds of Theorem~\ref{thm:experimental-matching}. Then, with
probability at least $1 - C e^{-c s}$,
\begin{equation*}
        \sup_{S \subseteq [p],\, |S| = s} \hat \beta_{0, S} - \beta_0
        \;\asymp\;
        \frac{\sqrt{\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2}}{\sigma_0}\,
        \frac{s \log(ep/s)}{n},
\end{equation*}
the centering at $\beta_0$ being legitimate because the sampling noise common
to all specifications, of order $1/\sqrt n$, is dominated at this scaling.}
\end{corollary}

\begin{remark}[Role of the distributional assumptions]\label{rem:subgaussian-upper}
\tc{The upper bound \eqref{eq:experimental-upper} rests on entropy and
chaining arguments, which consume only the subgaussian tails of
Assumption~\ref{ass:dgp}(i) and the proxy-free energy bound of condition
(v). The lower bound \eqref{eq:experimental-lower}
uses in addition anti-concentration: the independence of
Assumption~\ref{ass:experimental}, the treatment variance floor
$\sigma_0^2 > 0$, and the retained outcome variance of condition (iv) provide
the required small-ball estimates for the quadratic forms involved, so
Gaussianity is required nowhere. The moderate-deviations condition
$\log(ep/s) \lesssim \sqrt n$ is the price of this generality: it keeps the
searched statistics in the range where subgaussian tails match Gaussian
ones.}
\end{remark}

\begin{remark}[Sparsely explainable outcomes]\label{rem:sparse-proxy}
\tc{Without condition (v), both bounds of
Theorem~\ref{thm:experimental-matching} remain sharp after replacing
$\beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2$ by
\begin{equation*}
        \beta_{-0}^\top C \beta_{-0} + \sigma_\varepsilon^2
        + \frac{n}{s \log(ep/s)}
        \max_{|J| \le s}\, (C\beta_{-0})_J^\top C_{JJ}^{-1} (C\beta_{-0})_J:
\end{equation*}
a sparsely explainable outcome is strictly more hackable, and the inflation
is governed by the outcome variance that the best $s$ controls explain, not
by the conditioning of $C$.}
\end{remark}
